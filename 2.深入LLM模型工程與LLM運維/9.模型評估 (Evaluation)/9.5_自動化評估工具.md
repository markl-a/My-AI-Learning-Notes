#### 2. 自定義評估邏輯

```python
# custom_eval.py
import evals
from evals.api import CompletionFn
from evals.elsuite import utils

class CustomEval(evals.Eval):
    def __init__(self, completion_fns, *args, **kwargs):
        super().__init__(completion_fns, *args, **kwargs)
        self.samples = utils.get_jsonl(self.samples_jsonl)

    def eval_sample(self, sample, *_):
        """評估單個樣本"""
        prompt = sample["input"]
        ideal = sample["ideal"]

        # 獲取模型回答
        result = self.completion_fn(
            prompt=prompt,
            temperature=0.0,
            max_tokens=100
        )

        # 評估結果
        response = result.get_completions()[0]
        correct = self.check_answer(response, ideal)

        # 記錄結果
        evals.record.record_match(
            correct=correct,
            expected=ideal,
            picked=response
        )

        return {"accuracy": int(correct)}

    def check_answer(self, response, ideal):
        """檢查答案是否正確"""
        if isinstance(ideal, list):
            return any(i.lower() in response.lower() for i in ideal)
        return ideal.lower() in response.lower()

    def run(self, recorder):
        """運行評估"""
        samples = self.get_samples()
        self.eval_all_samples(recorder, samples)

        # 計算總體指標
        return {
            "accuracy": evals.metrics.get_accuracy(recorder.get_events())
        }
```

## 3. HELM (Holistic Evaluation of Language Models)

### 📋 基本信息

- **維護者**：Stanford CRFM
- **網址**：https://crfm.stanford.edu/helm/
- **特點**：全面、多維度評估框架
- **目標**：提供透明、標準化的評估

### 🎯 評估維度

HELM 從多個維度評估模型：

1. **準確性 (Accuracy)**：任務表現
2. **校準 (Calibration)**：置信度準確性
3. **魯棒性 (Robustness)**：抗干擾能力
4. **公平性 (Fairness)**：不同群體的表現
5. **偏見 (Bias)**：刻板印象和偏見
6. **毒性 (Toxicity)**：有害內容生成
7. **效率 (Efficiency)**：計算資源消耗

### 🔧 使用方法

```bash
# 安裝
pip install crfm-helm

# 運行評估
helm-run \
    --suite v1 \
    --models-to-run openai/gpt-3.5-turbo \
    --output-path benchmark_output

# 生成報告
helm-summarize \
    --suite v1 \
    --output-path benchmark_output
```

### 🔧 自定義評估器

```python
from langchain.evaluation import StringEvaluator

class CustomFactChecker(StringEvaluator):
    """自定義事實檢查評估器"""

    def __init__(self, fact_database):
        self.fact_database = fact_database

    def _evaluate_strings(
        self,
        prediction: str,
        reference: str = None,
        input: str = None,
        **kwargs
    ) -> dict:
        # 檢查預測中的事實
        facts_mentioned = self.extract_facts(prediction)
        correct_facts = sum(
            1 for fact in facts_mentioned
            if self.verify_fact(fact)
        )

        accuracy = correct_facts / len(facts_mentioned) if facts_mentioned else 0

        return {
            "score": accuracy,
            "details": {
                "total_facts": len(facts_mentioned),
                "correct_facts": correct_facts
            }
        }

    def extract_facts(self, text):
        # 提取事實（簡化版）
        return text.split(".")

    def verify_fact(self, fact):
        # 驗證事實
        return fact.strip() in self.fact_database
```

## 5. LLM-as-a-Judge（使用 LLM 評估）

### 📋 概念

使用強大的 LLM（如 GPT-4）作為評判者，評估其他模型的輸出質量。

### 🎯 優勢與局限

**優勢：**
- 捕捉細微的質量差異
- 評估開放式生成任務
- 快速、可擴展
- 與人類判斷相關性高

**局限性：**
- 可能有偏見（偏好冗長、特定風格）
- 成本較高（API 調用費用）
- "自我偏好"問題（偏好自己的輸出）
- 不適合需要專業知識的評估

### 🔧 實現方法

#### 1. 簡單評分

```python
from openai import OpenAI

client = OpenAI()

def llm_judge(question, answer, criteria="helpfulness"):
    """使用 GPT-4 評估答案質量"""

    prompt = f"""請評估以下回答的質量（評分 1-10）。

評估標準：{criteria}

問題：{question}

回答：{answer}

請給出分數（1-10）和簡短理由。格式：
分數：[數字]
理由：[說明]
"""

    response = client.chat.completions.create(
        model="gpt-4",
        messages=[{"role": "user", "content": prompt}],
        temperature=0.0
    )

    result = response.choices[0].message.content

    # 解析結果
    score = int(result.split("分數：")[1].split("\n")[0])
    reason = result.split("理由：")[1]

    return {"score": score, "reason": reason}

# 使用示例
evaluation = llm_judge(
    question="什麼是量子計算？",
    answer="量子計算是利用量子力學現象進行計算的技術...",
    criteria="準確性和清晰度"
)

print(f"分數：{evaluation['score']}/10")
print(f"理由：{evaluation['reason']}")
```

#### 2. 成對比較

```python
def llm_compare(question, answer_a, answer_b):

問題：{question}

回答 A：{answer_a}

回答 B：{answer_b}

評估標準：
1. 準確性
2. 完整性
3. 清晰度
4. 實用性

請選擇 A 或 B，並說明理由。格式：
選擇：[A/B/平手]
理由：[說明]
"""

    response = client.chat.completions.create(
        model="gpt-4",
        messages=[{"role": "user", "content": prompt}],
        temperature=0.0

    choice = result.split("選擇：")[1].split("\n")[0].strip()
    reason = result.split("理由：")[1]

    return {"winner": choice, "reason": reason}
```

## 6. 評估框架對比

| 框架 | 最適合 | 優勢 | 劣勢 |
|------|--------|------|------|
| **LM Eval Harness** | 學術研究、開源模型 | 任務最全、社群活躍 | 配置稍複雜 |
| **OpenAI Evals** | OpenAI 模型、快速原型 | 簡單易用、模組化 | 主要支持 OpenAI 模型 |
| **HELM** | 全面評估、研究報告 | 多維度、透明 | 運行耗時、成本高 |
| **LangChain** | LLM 應用鏈 | 應用層評估、靈活 | 較新、功能還在擴充 |
| **LLM-as-Judge** | 開放式生成任務 | 靈活、與人類相關性高 | 有偏見、成本高 |

### 評估腳本

```python
# scripts/auto_evaluate.py
import json
from lm_eval import evaluator
from datetime import datetime

class AutoEvaluator:
    def __init__(self, model_path, tasks):
        self.model_path = model_path
        self.tasks = tasks
        self.results = {}

    def run_evaluation(self):
        """運行評估"""
        print(f"開始評估：{self.model_path}")
        print(f"任務：{', '.join(self.tasks)}")

        results = evaluator.simple_evaluate(
            model="hf",
            model_args=f"pretrained={self.model_path}",
            tasks=self.tasks,
            num_fewshot=5,
            batch_size=8
        )

        self.results = results
        return results

    def check_thresholds(self, thresholds):
        """檢查是否達到性能閾值"""
        passed = True
        for task, min_score in thresholds.items():
            actual_score = self.results['results'][task]['acc']
            if actual_score < min_score:
                print(f"❌ {task}: {actual_score:.2%} < {min_score:.2%}")
                passed = False
            else:
                print(f"✅ {task}: {actual_score:.2%} >= {min_score:.2%}")

        return passed

    def save_results(self, output_path):
        """保存結果"""
        output = {
            "timestamp": datetime.now().isoformat(),
            "model": self.model_path,
            "tasks": self.tasks,
            "results": self.results
        }

        with open(output_path, 'w') as f:
            json.dump(output, f, indent=2)

        print(f"結果已保存到：{output_path}")

# 使用示例
if __name__ == "__main__":
    evaluator = AutoEvaluator(
        model_path="./my-model",
        tasks=["mmlu", "hellaswag", "arc_challenge"]
    )

    evaluator.run_evaluation()

    thresholds = {
        "mmlu": 0.60,
        "hellaswag": 0.75,
        "arc_challenge": 0.50
    }

    passed = evaluator.check_thresholds(thresholds)
    evaluator.save_results("eval_results.json")

    if not passed:
        exit(1)  # CI/CD 失敗
```
