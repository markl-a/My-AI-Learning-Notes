## 1. 建立多層次評估體系

### 🎯 三層評估金字塔

```
              ┌─────────────────┐
              │   人類評估      │ ← 最終驗證（樣本抽檢）
              │ 成本高、最準確   │
              └─────────────────┘
                      ▲
              ┌───────────────────┐
              │   LLM-as-Judge   │ ← 中間層（大規模評估）
              │   快速、可擴展    │
              └───────────────────┘
                      ▲
        ┌─────────────────────────────┐
        │   自動化指標（BLEU, Acc）   │ ← 基礎監控（持續運行）
        │   快速、低成本、粗略         │
        └─────────────────────────────┘
```

### 📋 實施策略

```python
class MultiLevelEvaluation:
    """多層次評估系統"""

    def __init__(self, model):
        self.model = model
        self.results = {}

    def level_1_automated(self, test_set):
        """第一層：自動化指標（全量數據）"""
        print("🔵 執行第一層評估：自動化指標")

        from lm_eval import evaluator

        results = evaluator.simple_evaluate(
            model=self.model,
            tasks=["mmlu", "hellaswag", "arc_challenge"],
            num_fewshot=5
        )

        self.results['automated'] = results
        return results

    def level_2_llm_judge(self, test_set, sample_size=500):
        """第二層：LLM 評判（中等樣本）"""
        print(f"🟡 執行第二層評估：LLM 評判（{sample_size} 樣本）")

        import random
        samples = random.sample(test_set, sample_size)

        scores = []
        for sample in samples:
            response = self.model.generate(sample['prompt'])
            score = self.llm_judge(sample['prompt'], response)
            scores.append(score)

        avg_score = sum(scores) / len(scores)
        self.results['llm_judge'] = avg_score

        return avg_score

    def level_3_human(self, test_set, sample_size=100):
        """第三層：人類評估（小樣本）"""
        print(f"🟢 執行第三層評估：人類評估（{sample_size} 樣本）")

        import random
        samples = random.sample(test_set, sample_size)

        # 發送給人類評估者
        human_eval_results = self.send_to_human_evaluators(samples)

        self.results['human'] = human_eval_results
        return human_eval_results

    def generate_report(self):
        """生成綜合評估報告"""
        return {
            "automated_metrics": self.results.get('automated'),
            "llm_judge_score": self.results.get('llm_judge'),
            "human_evaluation": self.results.get('human'),
            "recommendation": self.make_recommendation()
        }

    def make_recommendation(self):
        """基於評估結果給出建議"""
        if self.results.get('human', 0) > 0.8:
            return "✅ 模型準備就緒，可以部署"
        elif self.results.get('llm_judge', 0) > 0.7:
            return "⚠️ 需要進一步人類驗證"
        else:
            return "❌ 需要改進模型"
```

class TestSetDesign:
    """科學的測試集設計"""

    @staticmethod
    def split_data(data, train_ratio=0.8, val_ratio=0.1, test_ratio=0.1):
        """按時間切分，避免數據洩漏"""
        # 按時間戳排序
        data_sorted = sorted(data, key=lambda x: x['timestamp'])

        n = len(data_sorted)
        train_end = int(n * train_ratio)
        val_end = int(n * (train_ratio + val_ratio))

        return {
            'train': data_sorted[:train_end],
            'validation': data_sorted[train_end:val_end],
            'test': data_sorted[val_end:]
        }

    @staticmethod
    def stratified_split(data, labels, test_size=0.1):
        """分層抽樣，保持類別分布"""
        from sklearn.model_selection import train_test_split

        train_data, test_data = train_test_split(
            data,
            test_size=test_size,
            stratify=labels,
            random_state=42
        )

        return train_data, test_data

    @staticmethod
    def ensure_diversity(test_set):
        """確保測試集多樣性"""
        diversity_metrics = {
            'num_samples': len(test_set),
            'unique_topics': len(set(s['topic'] for s in test_set)),
            'difficulty_distribution': {},
            'length_distribution': {}
        }

        # 統計難度分布
        difficulties = [s.get('difficulty', 'unknown') for s in test_set]
        for diff in set(difficulties):
            diversity_metrics['difficulty_distribution'][diff] = \
                difficulties.count(diff) / len(difficulties)

        return diversity_metrics
```

class HoldoutSetManager:
    """管理保留集，防止過度優化測試集"""

    def __init__(self, secret_test_set):
        """保留集僅在最終評估時使用"""
        self._secret_test_set = secret_test_set
        self._evaluation_count = 0
        self.MAX_EVALUATIONS = 3  # 最多評估3次

    def can_evaluate(self):
        """檢查是否還能評估"""
        return self._evaluation_count < self.MAX_EVALUATIONS

    def evaluate(self, model):
        """在保留集上評估"""
        if not self.can_evaluate():
            raise ValueError("保留集評估次數已用完！")

        print(f"⚠️ 使用保留集評估（剩餘次數：{self.MAX_EVALUATIONS - self._evaluation_count - 1}）")

        # 執行評估
        results = self._run_evaluation(model, self._secret_test_set)
        self._evaluation_count += 1

        return results
```

```python
class ABTestingFramework:
    """A/B 測試框架"""

    def __init__(self, model_a, model_b, traffic_split=0.5):
        self.model_a = model_a
        self.model_b = model_b
        self.traffic_split = traffic_split
        self.results = {'A': [], 'B': []}

    def route_request(self, user_id):
        """根據用戶 ID 分配流量"""
        import hashlib

        hash_val = int(hashlib.md5(user_id.encode()).hexdigest(), 16)
        if (hash_val % 100) < (self.traffic_split * 100):
            return 'A'
        return 'B'

    def handle_request(self, user_id, query):
        """處理用戶請求"""
        variant = self.route_request(user_id)

        if variant == 'A':
            response = self.model_a.generate(query)
        else:
            response = self.model_b.generate(query)

        # 記錄結果
        self.log_interaction(variant, query, response)

        return response

    def log_interaction(self, variant, query, response):
        """記錄交互數據"""
        self.results[variant].append({
            'query': query,
            'response': response,
            'timestamp': datetime.now()
        })

    def analyze_results(self):
        """分析 A/B 測試結果"""
        metrics_a = self.calculate_metrics(self.results['A'])
        metrics_b = self.calculate_metrics(self.results['B'])

        # 統計顯著性檢驗
        p_value = self.significance_test(metrics_a, metrics_b)

        return {
            'variant_a': metrics_a,
            'variant_b': metrics_b,
            'p_value': p_value,
            'winner': self.determine_winner(metrics_a, metrics_b, p_value)
        }

    def significance_test(self, metrics_a, metrics_b):
        """統計顯著性檢驗"""
        from scipy import stats

        # 假設使用點擊率作為指標
        clicks_a = metrics_a.get('clicks', [])
        clicks_b = metrics_b.get('clicks', [])

        t_stat, p_value = stats.ttest_ind(clicks_a, clicks_b)
        return p_value

    def determine_winner(self, metrics_a, metrics_b, p_value):
        """確定勝者"""
        if p_value > 0.05:
            return "無顯著差異"
        elif metrics_a['mean'] > metrics_b['mean']:
            return "A"
        else:
            return "B"
```

#### 2. 影子模式（Shadow Mode）

```python
class ShadowMode:
    """影子模式：新模型與舊模型並行運行，但只返回舊模型結果"""

    def __init__(self, production_model, shadow_model):
        self.production_model = production_model
        self.shadow_model = shadow_model
        self.comparison_data = []

    def handle_request(self, query):
        """處理請求"""
        # 兩個模型並行運行
        import concurrent.futures

        with concurrent.futures.ThreadPoolExecutor() as executor:
            future_prod = executor.submit(self.production_model.generate, query)
            future_shadow = executor.submit(self.shadow_model.generate, query)

            prod_response = future_prod.result()
            shadow_response = future_shadow.result()

        # 只返回生產模型結果
        self.log_comparison(query, prod_response, shadow_response)

        return prod_response

    def log_comparison(self, query, prod_response, shadow_response):
        """記錄對比數據"""
        self.comparison_data.append({
            'query': query,
            'production': prod_response,
            'shadow': shadow_response,
            'timestamp': datetime.now()
        })

    def analyze_shadow_performance(self):
        """分析影子模型表現"""
        # 離線評估影子模型與生產模型的差異
        differences = []

        for record in self.comparison_data:
            diff = self.compare_responses(
                record['production'],
                record['shadow']
            )
            differences.append(diff)

        return {
            'average_difference': sum(differences) / len(differences),
            'ready_for_production': self.is_ready_for_production(differences)
        }

    def is_ready_for_production(self, differences):
        """判斷是否準備好部署"""
        # 如果 95% 的案例影子模型表現相當或更好
        good_cases = sum(1 for d in differences if d >= 0)
        return (good_cases / len(differences)) >= 0.95
```

class ContinuousEvaluation:
    """持續評估系統"""

    def __init__(self, model, alert_threshold=0.7):
        self.model = model
        self.alert_threshold = alert_threshold
        self.metrics_buffer = []
        self.alerts = []

    def monitor_request(self, query, response, user_feedback=None):
        """監控單次請求"""
        metrics = {
            'timestamp': datetime.now(),
            'latency': self.measure_latency(query, response),
            'token_count': len(response.split()),
            'refusal': self.is_refusal(response),
            'user_feedback': user_feedback
        }

        self.metrics_buffer.append(metrics)

        # 每100個請求分析一次
        if len(self.metrics_buffer) >= 100:
            self.analyze_and_alert()

    def analyze_and_alert(self):
        """分析並告警"""
        recent_metrics = self.metrics_buffer[-100:]

        # 計算關鍵指標
        avg_latency = sum(m['latency'] for m in recent_metrics) / 100
        refusal_rate = sum(m['refusal'] for m in recent_metrics) / 100
        satisfaction = self.calculate_satisfaction(recent_metrics)

        # 檢查閾值
        if satisfaction < self.alert_threshold:
            self.create_alert(
                level="WARNING",
                message=f"用戶滿意度下降：{satisfaction:.2%}",
                metrics={'satisfaction': satisfaction}
            )

        if refusal_rate > 0.1:  # 拒絕率 > 10%
            self.create_alert(
                level="WARNING",

class DataDriftDetector:
    """檢測輸入數據分布漂移"""

    def __init__(self, baseline_data):
        self.baseline_distribution = self.calculate_distribution(baseline_data)

    def calculate_distribution(self, data):
        """計算數據分布特徵"""
        from collections import Counter

        features = {
            'avg_length': sum(len(d) for d in data) / len(data),
            'topic_distribution': Counter(self.extract_topic(d) for d in data),
            'sentiment_distribution': Counter(self.extract_sentiment(d) for d in data)
        }

        return features

    def detect_drift(self, new_data):
        """檢測漂移"""
        new_distribution = self.calculate_distribution(new_data)

        # 計算分布差異
        drift_score = self.calculate_kl_divergence(
            self.baseline_distribution,
            new_distribution
        )

        if drift_score > 0.1:  # 閾值
            return {
                'drift_detected': True,
                'drift_score': drift_score,
                'recommendation': '建議重新評估模型'
            }

        return {'drift_detected': False}

## 🎯 總結

建立科學的評估體系需要：

### ✅ 核心原則

1. **多維度評估**：不依賴單一指標
2. **分層驗證**：自動化 + LLM + 人類
3. **持續監控**：不僅在部署前評估
4. **統計嚴謹**：計算顯著性，避免過度解讀
5. **目標對齊**：評估指標與業務目標一致

### 🔄 完整流程

```
1. 設計評估方案
   ↓
2. 準備測試數據（去重、分離）
   ↓
3. 執行離線評估（自動化指標）
   ↓
4. LLM-as-Judge 評估
   ↓
5. 人類評估（抽樣）
   ↓
6. 統計分析和可視化
   ↓
7. 決策（改進/部署）
   ↓
8. 部署後持續監控
   ↓
9. 定期重新評估
```
