### 效能對比

```python
import time
import torch
from transformers import AutoModelForCausalLM, AutoTokenizer

def benchmark_inference(model, tokenizer, prompt, num_runs=10):
    """基準測試推理速度"""
    input_ids = tokenizer(prompt, return_tensors="pt").input_ids.cuda()

    # 預熱
    for _ in range(3):
        with torch.no_grad():
            model.generate(input_ids, max_new_tokens=100)

    # 測試
    start = time.time()
    for _ in range(num_runs):
        with torch.no_grad():
            outputs = model.generate(
                input_ids,
                max_new_tokens=100,
                do_sample=True,
                temperature=0.7
            )
    end = time.time()

    avg_time = (end - start) / num_runs
    tokens_per_second = 100 / avg_time

    return {
        "avg_latency_ms": avg_time * 1000,
        "tokens_per_second": tokens_per_second
    }

# 測試原生 PyTorch 模型
model_pt = AutoModelForCausalLM.from_pretrained(
    "meta-llama/Llama-2-7b-hf",
    torch_dtype=torch.float16,
    device_map="auto"
)
tokenizer = AutoTokenizer.from_pretrained("meta-llama/Llama-2-7b-hf")

pt_result = benchmark_inference(model_pt, tokenizer, "Hello world")
print(f"PyTorch: {pt_result['tokens_per_second']:.2f} tokens/s")

# TensorRT 通常可以達到 2-4x 加速
print(f"預期 TensorRT: {pt_result['tokens_per_second'] * 3:.2f} tokens/s")
```

### PagedAttention 原理

```python
# PagedAttention 將 KV Cache 分頁管理
# 傳統方法：每個序列需要連續記憶體，容易碎片化
# PagedAttention：像虛擬記憶體一樣管理，減少浪費

"""
傳統 KV Cache 記憶體使用：
序列1: [████████░░] 80% 使用，但預留 100%
序列2: [███░░░░░░░] 30% 使用，但預留 100%
總浪費: ~50%

PagedAttention:
序列1: [████][████]       # 2 頁，完全使用
序列2: [███]               # 1 頁，部分使用
總浪費: ~10%
吞吐量提升: 2-4x
"""
```

## FastAPI 自建服務

### 基礎服務架構

```python
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
from transformers import AutoModelForCausalLM, AutoTokenizer
import torch
from typing import Optional, List
import uvicorn

app = FastAPI(title="LLM Inference API")

# 全局變數
model = None
tokenizer = None

class GenerateRequest(BaseModel):
    prompt: str
    max_tokens: int = 512
    temperature: float = 0.7
    top_p: float = 0.9
    top_k: int = 50
    repetition_penalty: float = 1.1
    stop_sequences: Optional[List[str]] = None

class GenerateResponse(BaseModel):
    generated_text: str
    tokens_generated: int
    prompt_tokens: int

@app.on_event("startup")
async def load_model():
    """啟動時載入模型"""
    global model, tokenizer
    print("Loading model...")

    model_name = "meta-llama/Llama-2-7b-chat-hf"

    tokenizer = AutoTokenizer.from_pretrained(model_name)
    model = AutoModelForCausalLM.from_pretrained(
        model_name,
        torch_dtype=torch.float16,
        device_map="auto",
        low_cpu_mem_usage=True
    )

    model.eval()
    print("Model loaded successfully!")

@app.post("/generate", response_model=GenerateResponse)
async def generate(request: GenerateRequest):
    """生成文本"""
    try:
        # Tokenize
        inputs = tokenizer(
            request.prompt,
            return_tensors="pt",
            truncation=True,
            max_length=2048
        ).to(model.device)

        prompt_tokens = inputs.input_ids.shape[1]

        # 生成
        with torch.no_grad():
            outputs = model.generate(
                inputs.input_ids,
                max_new_tokens=request.max_tokens,
                temperature=request.temperature,
                top_p=request.top_p,
                top_k=request.top_k,
                repetition_penalty=request.repetition_penalty,
                do_sample=True,
                pad_token_id=tokenizer.eos_token_id
            )

        # Decode
        generated_text = tokenizer.decode(
            outputs[0][prompt_tokens:],
            skip_special_tokens=True
        )

        tokens_generated = outputs.shape[1] - prompt_tokens

        return GenerateResponse(
            generated_text=generated_text,
            tokens_generated=tokens_generated,
            prompt_tokens=prompt_tokens
        )

    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@app.get("/health")
async def health_check():
    """健康檢查"""
    return {
        "status": "healthy",
        "model_loaded": model is not None
    }

if __name__ == "__main__":
    uvicorn.run(app, host="0.0.0.0", port=8000, workers=1)
```

### 添加串流支持

```python
from fastapi.responses import StreamingResponse
import json

@app.post("/generate/stream")
async def generate_stream(request: GenerateRequest):
    """串流生成文本"""
    async def event_generator():
        inputs = tokenizer(
            request.prompt,
            return_tensors="pt"
        ).to(model.device)

        # 使用 TextIteratorStreamer
        from transformers import TextIteratorStreamer
        from threading import Thread

        streamer = TextIteratorStreamer(
            tokenizer,
            skip_prompt=True,
            skip_special_tokens=True
        )

        generation_kwargs = dict(
            inputs=inputs.input_ids,
            max_new_tokens=request.max_tokens,
            temperature=request.temperature,
            top_p=request.top_p,
            do_sample=True,
            streamer=streamer
        )

        # 在後台執行緒中生成
        thread = Thread(target=model.generate, kwargs=generation_kwargs)
        thread.start()

        # 串流輸出
        for text in streamer:
            data = {
                "text": text,
                "done": False
            }
            yield f"data: {json.dumps(data)}\n\n"

        yield f"data: {json.dumps({'done': True})}\n\n"

    return StreamingResponse(
        event_generator(),
        media_type="text/event-stream"
    )
```

## 效能優化技術

### 1. 動態批次處理

```python
import asyncio
from collections import deque
from typing import List

class DynamicBatcher:
    def __init__(self, max_batch_size=8, max_wait_ms=50):
        self.max_batch_size = max_batch_size
        self.max_wait_ms = max_wait_ms
        self.queue = deque()
        self.lock = asyncio.Lock()

    async def add_request(self, request):
        """添加請求到批次隊列"""
        future = asyncio.Future()
        async with self.lock:
            self.queue.append((request, future))
        return await future

    async def process_batch(self, model, tokenizer):
        """處理一個批次"""
        while True:
            # 等待收集請求
            await asyncio.sleep(self.max_wait_ms / 1000)

            async with self.lock:
                if not self.queue:
                    continue

                # 取出批次
                batch_size = min(len(self.queue), self.max_batch_size)
                batch = [self.queue.popleft() for _ in range(batch_size)]

            # 批次推理
            requests, futures = zip(*batch)
            prompts = [r.prompt for r in requests]

            # Tokenize
            inputs = tokenizer(
                prompts,
                return_tensors="pt",
                padding=True,
                truncation=True
            ).to(model.device)

            # 生成
            with torch.no_grad():
                outputs = model.generate(
                    inputs.input_ids,
                    max_new_tokens=512,
                    temperature=0.7,
                    do_sample=True,
                    pad_token_id=tokenizer.eos_token_id
                )

            # 解碼並返回結果
            for i, future in enumerate(futures):
                generated_text = tokenizer.decode(
                    outputs[i],
                    skip_special_tokens=True
                )
                future.set_result(generated_text)

# 在 FastAPI 中使用
batcher = DynamicBatcher(max_batch_size=8, max_wait_ms=50)

@app.on_event("startup")
async def start_batcher():
    asyncio.create_task(batcher.process_batch(model, tokenizer))

@app.post("/generate")
async def generate(request: GenerateRequest):
    result = await batcher.add_request(request)
    return {"generated_text": result}
```

### 2. KV Cache 重用

# 使用 transformers 的 Cache 功能
from transformers import DynamicCache

def generate_with_cache(
    model, tokenizer, prompts, past_key_values=None
):
    """使用 KV Cache 加速生成"""
    inputs = tokenizer(prompts, return_tensors="pt").to(model.device)

    # 第一次生成，創建 Cache
    if past_key_values is None:
        outputs = model.generate(
            inputs.input_ids,
            max_new_tokens=50,
            return_dict_in_generate=True,
            output_hidden_states=False,
            use_cache=True
        )
        past_key_values = outputs.past_key_values
    else:
        # 重用 Cache（適合多輪對話）
        outputs = model.generate(
            inputs.input_ids,
            max_new_tokens=50,
            past_key_values=past_key_values,
            use_cache=True
        )

    return outputs, past_key_values

# 多輪對話範例
cache = None
for prompt in ["Hello", "How are you?", "Tell me more"]:
    outputs, cache = generate_with_cache(model, tokenizer, prompt, cache)
    print(tokenizer.decode(outputs.sequences[0]))

### 3. 推測性解碼（Speculative Decoding）

```python
# 使用小模型加速大模型推理
from transformers import AutoModelForCausalLM

# 載入大模型和小模型
large_model = AutoModelForCausalLM.from_pretrained(
    "meta-llama/Llama-2-70b-hf",
    device_map="auto",
    torch_dtype=torch.float16
)

small_model = AutoModelForCausalLM.from_pretrained(
    "meta-llama/Llama-2-7b-hf",
    device_map="auto",
    torch_dtype=torch.float16
)

# 推測性解碼
def speculative_decode(
    large_model, small_model, tokenizer, prompt, k=5
):
    """
    推測性解碼：
    1. 小模型快速生成 k 個 token
    2. 大模型驗證這些 token
    3. 保留正確的 token，丟棄錯誤的
    4. 平均加速 2-3x
    """
    input_ids = tokenizer(prompt, return_tensors="pt").input_ids

    generated = input_ids
    while generated.shape[1] < 512:
        # 小模型生成 k 個候選 token
        with torch.no_grad():
            small_outputs = small_model.generate(
                generated,
                max_new_tokens=k,
                do_sample=False
            )

        # 大模型驗證
        with torch.no_grad():
            large_logits = large_model(small_outputs).logits
            large_tokens = large_logits.argmax(dim=-1)

        # 找到第一個不匹配的位置
        matches = (small_outputs[:, -k:] == large_tokens[:, -k-1:-1])
        first_mismatch = (~matches).nonzero(as_tuple=True)[1]

        if len(first_mismatch) > 0:
            # 保留匹配的部分
            keep_tokens = first_mismatch[0].item()
            generated = small_outputs[:, :-(k-keep_tokens)]
        else:
            # 全部匹配，保留所有
            generated = small_outputs

    return tokenizer.decode(generated[0], skip_special_tokens=True)
```
