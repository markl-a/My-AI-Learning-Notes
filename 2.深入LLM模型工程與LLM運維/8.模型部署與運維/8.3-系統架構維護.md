```nginx
# nginx.conf
upstream llm_backend {
    # 負載均衡策略
    least_conn;  # 最少連接數（推薦用於 LLM，因為請求處理時間差異大）

    # 後端服務器
    server llm-server-1:8000 weight=3 max_fails=3 fail_timeout=30s;
    server llm-server-2:8000 weight=3 max_fails=3 fail_timeout=30s;
    server llm-server-3:8000 weight=2 max_fails=3 fail_timeout=30s;  # 較弱的機器

    # 備用服務器
    server llm-server-backup:8000 backup;

    # 保持連接活躍
    keepalive 32;
    keepalive_timeout 60s;
}

server {
    listen 80;
    server_name api.example.com;

    # 客戶端請求限制
    client_max_body_size 10M;
    client_body_timeout 60s;

    # 請求超時
    proxy_connect_timeout 60s;
    proxy_send_timeout 300s;
    proxy_read_timeout 300s;

    # 速率限制
    limit_req_zone $binary_remote_addr zone=api_limit:10m rate=10r/s;
    limit_req zone=api_limit burst=20 nodelay;

    location /v1/ {
        proxy_pass http://llm_backend;

        # Headers
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        # 串流支持
        proxy_buffering off;
        proxy_cache off;
        proxy_http_version 1.1;
        proxy_set_header Connection "";

        # 健康檢查
        proxy_next_upstream error timeout http_500 http_502 http_503;
    }

    # 健康檢查端點
    location /health {
        proxy_pass http://llm_backend/health;
        access_log off;
    }

    # Metrics 端點
    location /metrics {
        proxy_pass http://llm_backend/metrics;
        # 限制訪問
        allow 10.0.0.0/8;
        deny all;
    }
```

### 3. 智能路由與流量分配

```python
# intelligent_router.py
from fastapi import FastAPI, Request
from typing import Dict, List
import httpx
import hashlib
import time

app = FastAPI()

class LoadBalancer:
    def __init__(self):
        self.backends = [
            {"url": "http://backend-1:8000", "capacity": 100, "current": 0},
            {"url": "http://backend-2:8000", "capacity": 100, "current": 0},
            {"url": "http://backend-3:8000", "capacity": 80, "current": 0},
        ]
        self.stats: Dict[str, dict] = {}

    async def get_backend(self, request_complexity: int) -> str:
        """
        智能選擇後端：
        - 考慮當前負載
        - 考慮請求複雜度
        - 避免過載
        """
        # 計算每個後端的負載率
        available = []
        for backend in self.backends:
            load_ratio = backend["current"] / backend["capacity"]
            if load_ratio < 0.9:  # 不超過 90% 負載
                score = (1 - load_ratio) * backend["capacity"]
                available.append((backend, score))

        if not available:
            # 所有後端都繁忙，選擇負載最低的
            backend = min(self.backends, key=lambda x: x["current"])
        else:
            # 選擇得分最高的
            backend = max(available, key=lambda x: x[1])[0]

        backend["current"] += request_complexity
        return backend["url"]

    async def release_backend(self, url: str, complexity: int):
        """釋放後端資源"""
        for backend in self.backends:
            if backend["url"] == url:
                backend["current"] = max(0, backend["current"] - complexity)
                break

    async def update_stats(self, backend_url: str, latency: float, success: bool):
        """更新統計信息"""
        if backend_url not in self.stats:
            self.stats[backend_url] = {
                "total_requests": 0,
                "successful_requests": 0,
                "total_latency": 0,
                "errors": 0
            }

        stats = self.stats[backend_url]
        stats["total_requests"] += 1
        stats["total_latency"] += latency

        if success:
            stats["successful_requests"] += 1
        else:
            stats["errors"] += 1

balancer = LoadBalancer()

@app.post("/v1/generate")
async def generate(request: Request):
    """帶智能路由的生成端點"""
    body = await request.json()
    prompt = body.get("prompt", "")

    # 估算請求複雜度
    complexity = len(prompt.split()) + body.get("max_tokens", 100)

    # 選擇後端
    backend_url = await balancer.get_backend(complexity)

    start_time = time.time()
    success = False

    try:
        # 轉發請求
        async with httpx.AsyncClient() as client:
            response = await client.post(
                f"{backend_url}/v1/generate",
                json=body,
                timeout=300.0
            )
            response.raise_for_status()
            success = True
            return response.json()

    except Exception as e:
        # 重試其他後端
        for backend in balancer.backends:
            if backend["url"] != backend_url:
                try:
                    async with httpx.AsyncClient() as client:
                        response = await client.post(
                            f"{backend['url']}/v1/generate",
                            json=body,
                            timeout=300.0
                        )
                        response.raise_for_status()
                        success = True
                        return response.json()
                except:
                    continue
        raise HTTPException(status_code=503, detail="All backends unavailable")

    finally:
        latency = time.time() - start_time
        await balancer.release_backend(backend_url, complexity)
        await balancer.update_stats(backend_url, latency, success)

@app.get("/admin/stats")
async def get_stats():
    """獲取負載均衡統計"""
    return {
        "backends": balancer.backends,
        "stats": balancer.stats
    }
```

```python
                key=lambda x: x[1]["created_at"],
                reverse=True

        self,
        model_name: str,

        if model_name not in self.models:
```

### 2. A/B 測試框架

```python
# ab_testing.py
from typing import Dict, List
import random
import time

class ABTestManager:
    def __init__(self):
        self.experiments: Dict[str, dict] = {}
        self.results: Dict[str, List[dict]] = {}

    def create_experiment(
        self,
        experiment_id: str,
        variants: Dict[str, dict],
        traffic_split: Dict[str, float]
    ):
        """
        創建 A/B 測試實驗

        Args:
            experiment_id: 實驗 ID
            variants: 變體配置 {"A": {"model": "v1.0"}, "B": {"model": "v1.1"}}
            traffic_split: 流量分配 {"A": 50.0, "B": 50.0}
        """
        if sum(traffic_split.values()) != 100.0:
            raise ValueError("Traffic split must sum to 100")

        self.experiments[experiment_id] = {
            "variants": variants,
            "traffic_split": traffic_split,
            "created_at": time.time(),
            "total_requests": 0
        }
        self.results[experiment_id] = []

    def assign_variant(self, experiment_id: str, user_id: str = None) -> str:
        """
        分配變體

        使用一致性雜湊確保同一用戶總是看到同一變體
        """
        experiment = self.experiments[experiment_id]

        if user_id:
            # 一致性雜湊
            hash_value = int(hashlib.md5(user_id.encode()).hexdigest(), 16)
            percentage = (hash_value % 100) / 100.0 * 100
        else:
            # 隨機分配
            percentage = random.random() * 100

        # 根據流量分配選擇變體
        cumulative = 0
        for variant, split in experiment["traffic_split"].items():
            cumulative += split
            if percentage <= cumulative:
                return variant

        # 默認返回第一個變體
        return list(experiment["variants"].keys())[0]

    def record_result(
        self,
        experiment_id: str,
        variant: str,
        metrics: dict
    ):
        """記錄實驗結果"""
        self.results[experiment_id].append({
            "variant": variant,
            "metrics": metrics,
            "timestamp": time.time()
        })
        self.experiments[experiment_id]["total_requests"] += 1

    def analyze_results(self, experiment_id: str) -> dict:
        """分析實驗結果"""
        results = self.results[experiment_id]
        variants = {}

        for result in results:
            variant = result["variant"]
            if variant not in variants:
                variants[variant] = {
                    "count": 0,
                    "total_latency": 0,
                    "total_tokens": 0,
                    "errors": 0
                }

            v = variants[variant]
            v["count"] += 1
            v["total_latency"] += result["metrics"].get("latency", 0)
            v["total_tokens"] += result["metrics"].get("tokens", 0)
            if result["metrics"].get("error"):
                v["errors"] += 1

        # 計算平均值
        summary = {}
        for variant, data in variants.items():
            summary[variant] = {
                "requests": data["count"],
                "avg_latency": data["total_latency"] / data["count"],
                "avg_tokens": data["total_tokens"] / data["count"],
                "error_rate": data["errors"] / data["count"] * 100,
            }

        return summary
```

```python
@app.middleware("http")
async def monitor_requests(request: Request, call_next):
    """監控中間件"""
    active_requests.inc()
    start_time = time.time()

    try:
        response = await call_next(request)
        latency = time.time() - start_time

        # 記錄指標
        request_count.labels(
            model=request.state.model_name,
            endpoint=request.url.path,
            status=response.status_code
        ).inc()

        request_latency.labels(
            model=request.state.model_name,
            endpoint=request.url.path
        ).observe(latency)

        return response

    except Exception as e:
        request_count.labels(
            model="unknown",
            endpoint=request.url.path,
            status=500
        ).inc()
        raise

    finally:
        active_requests.dec()

@app.get("/metrics")

        await asyncio.sleep(10)  # 每 10 秒更新一次
```

```python
# custom_dashboard.py
from fastapi import FastAPI
from typing import Dict, List
import time
from collections import deque

class MetricsCollector:
    def __init__(self, max_history=1000):
        self.max_history = max_history
        self.metrics = {
            "requests": deque(maxlen=max_history),
            "latencies": deque(maxlen=max_history),
            "errors": deque(maxlen=max_history),
            "throughput": deque(maxlen=max_history)
        }
        self.start_time = time.time()

    def record_request(
        self,
        latency: float,
        tokens: int,
        error: bool = False
    ):
        """記錄請求"""
        timestamp = time.time()
        self.metrics["requests"].append({
            "timestamp": timestamp,
            "latency": latency,
            "tokens": tokens,
            "error": error
        })

    def get_stats(self, window_seconds: int = 60) -> dict:
        """獲取統計信息"""
        now = time.time()
        cutoff = now - window_seconds

        # 過濾時間窗口內的數據
        recent_requests = [
            r for r in self.metrics["requests"]
            if r["timestamp"] > cutoff
        ]

        if not recent_requests:
            return {
                "requests_per_second": 0,
                "avg_latency": 0,
                "error_rate": 0,
                "tokens_per_second": 0
            }

        total_requests = len(recent_requests)
        total_latency = sum(r["latency"] for r in recent_requests)
        total_errors = sum(1 for r in recent_requests if r["error"])
        total_tokens = sum(r["tokens"] for r in recent_requests)

        return {
            "requests_per_second": total_requests / window_seconds,
            "avg_latency": total_latency / total_requests,
            "p95_latency": self._percentile(
                [r["latency"] for r in recent_requests], 95
            ),
            "p99_latency": self._percentile(
                [r["latency"] for r in recent_requests], 99
            ),
            "error_rate": (total_errors / total_requests) * 100,
            "tokens_per_second": total_tokens / window_seconds,
            "total_requests": total_requests
        }

    def _percentile(self, data: List[float], percentile: int) -> float:
        """計算百分位數"""
        sorted_data = sorted(data)
        index = int(len(sorted_data) * percentile / 100)
        return sorted_data[min(index, len(sorted_data) - 1)]

metrics_collector = MetricsCollector()

@app.get("/dashboard/stats")
async def get_dashboard_stats(window: int = 60):
    """獲取儀表板統計數據"""
    return metrics_collector.get_stats(window)

@app.get("/dashboard/health")
async def health_dashboard():
    """綜合健康狀態"""
    stats = metrics_collector.get_stats(60)

    # 健康檢查邏輯
    health_status = "healthy"
    issues = []

    if stats["error_rate"] > 5.0:
        health_status = "degraded"
        issues.append(f"High error rate: {stats['error_rate']:.2f}%")

    if stats["avg_latency"] > 10.0:
        health_status = "degraded"
        issues.append(f"High latency: {stats['avg_latency']:.2f}s")

    if stats["error_rate"] > 20.0 or stats["avg_latency"] > 30.0:
        health_status = "unhealthy"

    return {
        "status": health_status,
        "stats": stats,
        "issues": issues,
        "timestamp": time.time()
    }
```

```python
# structured_logging.py
import logging
import json
from datetime import datetime
from typing import Any, Dict

class JSONFormatter(logging.Formatter):
    """JSON 格式化日誌"""
    def format(self, record: logging.LogRecord) -> str:
        log_data = {
            "timestamp": datetime.utcnow().isoformat(),
            "level": record.levelname,
            "logger": record.name,
            "message": record.getMessage(),
            "module": record.module,
            "function": record.funcName,
            "line": record.lineno
        }

        # 添加額外字段
        if hasattr(record, "request_id"):
            log_data["request_id"] = record.request_id
        if hasattr(record, "user_id"):
            log_data["user_id"] = record.user_id
        if hasattr(record, "model_name"):
            log_data["model_name"] = record.model_name

        # 添加異常信息
        if record.exc_info:
            log_data["exception"] = self.formatException(record.exc_info)

        return json.dumps(log_data)

# 配置日誌
def setup_logging(level=logging.INFO):
    """設置日誌配置"""
    logger = logging.getLogger()
    logger.setLevel(level)

    # 控制台處理器
    console_handler = logging.StreamHandler()
    console_handler.setFormatter(JSONFormatter())
    logger.addHandler(console_handler)

    # 文件處理器
    file_handler = logging.FileHandler("llm_api.log")
    file_handler.setFormatter(JSONFormatter())
    logger.addHandler(file_handler)

    return logger

logger = setup_logging()

# 在 FastAPI 中使用
@app.middleware("http")
async def logging_middleware(request: Request, call_next):
    """日誌中間件"""
    request_id = str(uuid.uuid4())
    request.state.request_id = request_id

    # 記錄請求
    logger.info(
        "Incoming request",
        extra={
            "request_id": request_id,
            "method": request.method,
            "path": request.url.path,
            "client_ip": request.client.host
        }
    )

    start_time = time.time()

    try:
        response = await call_next(request)
        latency = time.time() - start_time

        # 記錄響應
        logger.info(
            "Request completed",
            extra={
                "request_id": request_id,
                "status_code": response.status_code,
                "latency": latency
            }
        )

        return response

    except Exception as e:
        latency = time.time() - start_time
        logger.error(
            "Request failed",
            extra={
                "request_id": request_id,
                "latency": latency,
                "error": str(e)
            },
            exc_info=True
        )
        raise
```

```python
# diagnostic_tools.py
import psutil
import torch
import subprocess

class DiagnosticTools:
    """診斷工具集"""

    @staticmethod
    def check_gpu_health():
        """檢查 GPU 健康狀態"""
        if not torch.cuda.is_available():
            return {"status": "error", "message": "CUDA not available"}

        diagnostics = {
            "gpu_count": torch.cuda.device_count(),
            "gpus": []
        }

        for i in range(torch.cuda.device_count()):
            gpu_info = {
                "id": i,
                "name": torch.cuda.get_device_name(i),
                "memory_allocated": torch.cuda.memory_allocated(i) / 1024**3,
                "memory_reserved": torch.cuda.memory_reserved(i) / 1024**3,
                "memory_total": torch.cuda.get_device_properties(i).total_memory / 1024**3
            }

            # 計算使用率
            usage_percent = (gpu_info["memory_allocated"] /
                           gpu_info["memory_total"]) * 100
            gpu_info["usage_percent"] = usage_percent

            # 健康狀態
            if usage_percent > 95:
                gpu_info["status"] = "critical"
            elif usage_percent > 85:
                gpu_info["status"] = "warning"
            else:
                gpu_info["status"] = "healthy"

            diagnostics["gpus"].append(gpu_info)

        return diagnostics

    @staticmethod
    def check_system_resources():
        """檢查系統資源"""
        cpu_percent = psutil.cpu_percent(interval=1)
        memory = psutil.virtual_memory()

        return {
            "cpu": {
                "usage_percent": cpu_percent,
                "count": psutil.cpu_count(),
                "status": "warning" if cpu_percent > 80 else "healthy"
            },
            "memory": {
                "total_gb": memory.total / 1024**3,
                "available_gb": memory.available / 1024**3,
                "used_percent": memory.percent,
                "status": "warning" if memory.percent > 85 else "healthy"
            },
            "disk": {
                "usage_percent": psutil.disk_usage('/').percent,
                "status": "warning" if psutil.disk_usage('/').percent > 85 else "healthy"
            }
        }

    def run_inference_test():
        """運行推理測試"""

            start_time = time.time()

            # result = model.generate(test_prompt)

            latency = time.time() - start_time

            return {
                "status": "success",
                "latency": latency,
                "timestamp": time.time()
            }

        except Exception as e:
            return {
                "status": "failed",
                "error": str(e),
                "timestamp": time.time()
            }
```
