### 2. 靜態資料加密 (Encryption at Rest)

```python
# 使用 Fernet 對稱加密敏感資料
from cryptography.fernet import Fernet
import os
import json

class DataEncryption:
    def __init__(self, key_file="encryption.key"):
        self.key_file = key_file
        self.key = self._load_or_generate_key()
        self.cipher = Fernet(self.key)

    def _load_or_generate_key(self):
        """載入或生成加密金鑰"""
        if os.path.exists(self.key_file):
            with open(self.key_file, "rb") as f:
                return f.read()
        else:
            key = Fernet.generate_key()
            # 金鑰應該安全存儲（例如 AWS KMS、HashiCorp Vault）
            with open(self.key_file, "wb") as f:
                f.write(key)
            # 設置嚴格權限
            os.chmod(self.key_file, 0o600)
            return key

    def encrypt_data(self, data: dict) -> bytes:
        """加密資料"""
        json_data = json.dumps(data).encode()
        encrypted = self.cipher.encrypt(json_data)
        return encrypted

    def decrypt_data(self, encrypted_data: bytes) -> dict:
        """解密資料"""
        decrypted = self.cipher.decrypt(encrypted_data)
        return json.loads(decrypted.decode())

    def encrypt_file(self, input_file: str, output_file: str):
        """加密檔案"""
        with open(input_file, "rb") as f:
            data = f.read()

        encrypted_data = self.cipher.encrypt(data)

        with open(output_file, "wb") as f:
            f.write(encrypted_data)

        # 安全刪除原始檔案
        self._secure_delete(input_file)

    def _secure_delete(self, filename: str):
        """安全刪除檔案（多次覆寫）"""
        if os.path.exists(filename):
            # 覆寫 3 次
            with open(filename, "ba+") as f:
                length = f.tell()
                for _ in range(3):
                    f.seek(0)
                    f.write(os.urandom(length))
            os.remove(filename)

# 使用範例
encryptor = DataEncryption()

# 加密敏感用戶資料
user_data = {
    "user_id": "12345",
    "prompt": "My secret prompt",
    "response": "Generated response"
}

encrypted = encryptor.encrypt_data(user_data)
# 儲存加密資料到資料庫

# 解密
decrypted = encryptor.decrypt_data(encrypted)
```

```python
# api_key_auth.py
from fastapi import Security, HTTPException, status
from fastapi.security import APIKeyHeader
from typing import Optional
import secrets
import hashlib
from datetime import datetime, timedelta

API_KEY_HEADER = APIKeyHeader(name="X-API-Key", auto_error=False)

class APIKeyManager:
    def __init__(self):
        # 實際應該存儲在資料庫中
        self.api_keys = {}  # {hashed_key: {user_id, permissions, expires_at}}

    def generate_api_key(
        self,
        user_id: str,
        permissions: list = None,
        expires_in_days: int = 365
    ) -> str:
        """生成新的 API 金鑰"""
        # 生成強隨機金鑰
        api_key = f"sk-{secrets.token_urlsafe(32)}"

        # 雜湊金鑰存儲
        hashed_key = hashlib.sha256(api_key.encode()).hexdigest()

        self.api_keys[hashed_key] = {
            "user_id": user_id,
            "permissions": permissions or ["read", "write"],
            "created_at": datetime.utcnow(),
            "expires_at": datetime.utcnow() + timedelta(days=expires_in_days),
            "last_used": None,
            "usage_count": 0
        }

        return api_key

    def verify_api_key(self, api_key: str) -> Optional[dict]:
        """驗證 API 金鑰"""
        hashed_key = hashlib.sha256(api_key.encode()).hexdigest()

        if hashed_key not in self.api_keys:
            return None

        key_info = self.api_keys[hashed_key]

        # 檢查過期
        if datetime.utcnow() > key_info["expires_at"]:
            return None

        # 更新使用記錄
        key_info["last_used"] = datetime.utcnow()
        key_info["usage_count"] += 1

        return key_info

    def revoke_api_key(self, api_key: str):
        """撤銷 API 金鑰"""
        hashed_key = hashlib.sha256(api_key.encode()).hexdigest()
        if hashed_key in self.api_keys:
            del self.api_keys[hashed_key]
            return True
        return False

api_key_manager = APIKeyManager()

async def get_api_key(api_key: str = Security(API_KEY_HEADER)) -> dict:
    """API 金鑰驗證依賴"""
    if not api_key:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Missing API Key"
        )

    key_info = api_key_manager.verify_api_key(api_key)

    if not key_info:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid or expired API Key"
        )

    return key_info

# 使用範例

    if "generate" not in key_info["permissions"]:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="Insufficient permissions"
        )

    # 處理請求...
    pass

@app.post("/admin/api-keys")
async def create_api_key(
```

```python
    """驗證密碼"""
    return pwd_context.verify(plain_password, hashed_password)

def get_password_hash(password: str) -> str:
    """雜湊密碼"""
    return pwd_context.hash(password)

def authenticate_user(username: str, password: str) -> Optional[User]:
    """認證用戶"""
    user = users_db.get(username)
    if not user or not verify_password(password, user.hashed_password):
        return None
    return user

def create_access_token(
    data: dict,
    expires_delta: Optional[timedelta] = None
) -> str:
    """建立訪問 token"""
    to_encode = data.copy()

    if expires_delta:
        expire = datetime.utcnow() + expires_delta
    else:
        expire = datetime.utcnow() + timedelta(minutes=15)

    to_encode.update({
        "exp": expire,
        "iat": datetime.utcnow(),
        "type": "access"
    })

    encoded_jwt = jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)
    return encoded_jwt

def create_refresh_token(data: dict) -> str:
    """建立刷新 token"""
    to_encode = data.copy()
    expire = datetime.utcnow() + timedelta(days=REFRESH_TOKEN_EXPIRE_DAYS)

    to_encode.update({
        "exp": expire,
        "iat": datetime.utcnow(),
        "type": "refresh"
    })

    encoded_jwt = jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)
    return encoded_jwt

async def get_current_user(token: str = Depends(oauth2_scheme)) -> User:
    """獲取當前用戶"""
    credentials_exception = HTTPException(
        status_code=status.HTTP_401_UNAUTHORIZED,
        detail="Could not validate credentials",
        headers={"WWW-Authenticate": "Bearer"},
    )

    try:
        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
        username: str = payload.get("sub")

        if username is None or payload.get("type") != "access":
            raise credentials_exception

        # 檢查 token 是否過期
        exp = payload.get("exp")
        if datetime.fromtimestamp(exp) < datetime.utcnow():
            raise credentials_exception

    except JWTError:
        raise credentials_exception

    user = users_db.get(username)
    if user is None:
        raise credentials_exception

    return user

def require_roles(allowed_roles: list):
    """角色檢查裝飾器"""
    async def role_checker(current_user: User = Depends(get_current_user)):
        if not any(role in current_user.roles for role in allowed_roles):
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="Insufficient permissions"
            )
        return current_user
    return role_checker

# API 端點
```

```python
# rbac.py
from enum import Enum
from typing import List, Set

class Permission(str, Enum):
    """權限定義"""
    READ = "read"
    WRITE = "write"
    DELETE = "delete"
    ADMIN = "admin"
    GENERATE_TEXT = "generate_text"
    VIEW_LOGS = "view_logs"
    MANAGE_USERS = "manage_users"

class Role:
    """角色定義"""
    def __init__(self, name: str, permissions: Set[Permission]):
        self.name = name
        self.permissions = permissions

    def has_permission(self, permission: Permission) -> bool:
        return permission in self.permissions

# 預定義角色
ROLES = {
    "guest": Role("guest", {Permission.READ}),
    "user": Role("user", {
        Permission.READ,
        Permission.WRITE,
        Permission.GENERATE_TEXT
    }),
    "power_user": Role("power_user", {
        Permission.READ,
        Permission.WRITE,
        Permission.DELETE,
        Permission.GENERATE_TEXT,
        Permission.VIEW_LOGS
    }),
    "admin": Role("admin", {
        Permission.READ,
        Permission.WRITE,
        Permission.DELETE,
        Permission.ADMIN,
        Permission.GENERATE_TEXT,
        Permission.VIEW_LOGS,
        Permission.MANAGE_USERS
    })
}

def check_permission(user_role: str, required_permission: Permission) -> bool:
    """檢查用戶是否有特定權限"""
    role = ROLES.get(user_role)
    if not role:
        return False
    return role.has_permission(required_permission)

# FastAPI 權限檢查
from fastapi import Depends, HTTPException

def require_permission(permission: Permission):
    """權限檢查依賴"""
    async def permission_checker(

        has_perm = any(

        )

        if not has_perm:
            raise HTTPException(
                status_code=403,
                detail=f"Missing required permission: {permission}"
            )

        return current_user

    return permission_checker
```

```python
        # 台灣手機號碼

        # 信用卡號

    def detect_pii(self, text: str) -> List[Dict]:
        """偵測文字中的 PII"""
        results = self.analyzer.analyze(
            text=text,
            language="en",
            entities=[
                "PERSON",
                "EMAIL_ADDRESS",
                "PHONE_NUMBER",
                "CREDIT_CARD",
                "IBAN_CODE",
                "IP_ADDRESS",
                "LOCATION",
                "DATE_TIME"
            ]
        )

        for result in results:
            pii_found.append({
                "type": result.entity_type,
                "start": result.start,
                "end": result.end,
                "score": result.score,
                "text": text[result.start:result.end]
            })

        return pii_found

    def anonymize(
        self,
        text: str,
        anonymization_type: str = "replace"
    ) -> str:
        """
        匿名化文字中的 PII

        Args:
            text: 原始文字
            anonymization_type: replace, mask, hash, redact
        """
        # 分析
        analyzer_results = self.analyzer.analyze(
            text=text,
            language="en"
        )

        # 匿名化
        if anonymization_type == "replace":
            operators = {
                "PERSON": OperatorConfig("replace", {"new_value": "[PERSON]"}),
                "EMAIL_ADDRESS": OperatorConfig("replace", {"new_value": "[EMAIL]"}),
                "PHONE_NUMBER": OperatorConfig("replace", {"new_value": "[PHONE]"}),
                "CREDIT_CARD": OperatorConfig("replace", {"new_value": "[CARD]"}),
                "LOCATION": OperatorConfig("replace", {"new_value": "[LOCATION]"}),
            }
        elif anonymization_type == "mask":
            operators = {
                "DEFAULT": OperatorConfig("mask", {
                    "masking_char": "*",
                    "chars_to_mask": 100,
                    "from_end": False
                })
            }
        elif anonymization_type == "hash":
            operators = {
                "DEFAULT": OperatorConfig("hash", {"hash_type": "sha256"})
            }
        else:  # redact
            operators = {
                "DEFAULT": OperatorConfig("redact", {})
            }

        anonymized_result = self.anonymizer.anonymize(
            text=text,
            analyzer_results=analyzer_results,
            operators=operators
        )

        return anonymized_result.text

    def safe_log(self, text: str) -> str:
        """安全地記錄日誌（自動脫敏）"""
        return self.anonymize(text, anonymization_type="mask")
```

### 2. 差分隱私 (Differential Privacy)

```python
# differential_privacy.py
import numpy as np

class DifferentialPrivacy:
    """差分隱私實現"""

    @staticmethod
    def add_laplace_noise(
        value: float,
        sensitivity: float,
        epsilon: float
    ) -> float:
        """
        添加拉普拉斯噪音

        Args:
            value: 原始值
            sensitivity: 敏感度
            epsilon: 隱私預算（越小越私密）
        """
        scale = sensitivity / epsilon
        noise = np.random.laplace(0, scale)
        return value + noise

    @staticmethod
    def add_gaussian_noise(
        value: float,
        sensitivity: float,
        epsilon: float,
        delta: float
    ) -> float:
        """
        添加高斯噪音

        Args:
            delta: 失敗概率
        """
        sigma = sensitivity * np.sqrt(2 * np.log(1.25 / delta)) / epsilon
        noise = np.random.normal(0, sigma)
        return value + noise

    @staticmethod
    def privatize_statistics(
        data: np.ndarray,
        epsilon: float = 1.0
    ) -> dict:
        """
        私密化統計資料

        Args:
            data: 原始資料
            epsilon: 隱私預算
        """
        n = len(data)

        # 計算真實統計量
        true_mean = np.mean(data)
        true_std = np.std(data)

        # 添加噪音
        # 均值的敏感度：(max - min) / n
        data_range = np.max(data) - np.min(data)
        mean_sensitivity = data_range / n

        private_mean = DifferentialPrivacy.add_laplace_noise(
            true_mean,
            mean_sensitivity,
            epsilon / 2
        )

        # 標準差的敏感度
        std_sensitivity = data_range / n

        private_std = DifferentialPrivacy.add_laplace_noise(
            true_std,
            std_sensitivity,
            epsilon / 2
        )

        return {
            "mean": private_mean,
            "std": private_std,
            "count": n,  # 計數通常不加噪音，或加少量噪音
            "epsilon_used": epsilon
        }

class PrivateUsageStats:
```

```python
        self.clients.append(client)

    def aggregate_updates(
        self,
        client_updates: List[dict],
        aggregation_method: str = "fedavg"
    ):
        """
        聚合客戶端更新

        Args:
            aggregation_method: fedavg (Federated Averaging)
        """
        if aggregation_method == "fedavg":
            # 加權平均（按樣本數量）
            total_samples = sum(u["num_samples"] for u in client_updates)

            aggregated_params = {}
            for key in self.global_model.state_dict().keys():
                aggregated_params[key] = sum(
                    update["model_updates"][key] * update["num_samples"] / total_samples
                    for update in client_updates
                )

            self.global_model.load_state_dict(aggregated_params)

    def federated_round(self, num_epochs: int = 1):
        """執行一輪聯邦學習"""
        # 1. 分發全局模型到客戶端
        for client in self.clients:
            client.model.load_state_dict(
                self.global_model.state_dict()
            )

        # 2. 客戶端本地訓練
        client_updates = []
        for client in self.clients:
            update = client.train_local(epochs=num_epochs)
            client_updates.append(update)

        # 3. 聚合更新
        self.aggregate_updates(client_updates)

        return {
            "round_completed": True,
            "num_clients": len(client_updates),
            "total_samples": sum(u["num_samples"] for u in client_updates)
        }
```

```python
# gdpr_compliance.py
from datetime import datetime, timedelta
from typing import Optional

class GDPRCompliance:
    """GDPR 合規性管理"""

    def __init__(self):
        self.user_data = {}  # 實際應使用資料庫
        self.consent_records = {}
        self.data_retention_days = 365

    def record_consent(
        self,
        user_id: str,
        purpose: str,
        consent_given: bool
    ):
        """記錄用戶同意"""
        if user_id not in self.consent_records:
            self.consent_records[user_id] = []

        self.consent_records[user_id].append({
            "purpose": purpose,
            "consent": consent_given,
            "timestamp": datetime.utcnow(),
            "ip_address": None,  # 應記錄 IP
            "user_agent": None   # 應記錄 User-Agent
        })

    def has_consent(self, user_id: str, purpose: str) -> bool:
        """檢查是否有同意"""
        if user_id not in self.consent_records:
            return False

        # 獲取最新的同意記錄
        consents = [
            c for c in self.consent_records[user_id]
            if c["purpose"] == purpose
        ]

        if not consents:
            return False

        latest = max(consents, key=lambda x: x["timestamp"])
        return latest["consent"]

    def export_user_data(self, user_id: str) -> dict:
        """
        資料可攜權 (Right to Data Portability)
        GDPR Article 20
        """
        if user_id not in self.user_data:
            return {"error": "User not found"}

        # 收集所有用戶資料
        user_export = {
            "user_id": user_id,
            "exported_at": datetime.utcnow().isoformat(),
            "data": self.user_data[user_id],
            "consent_history": self.consent_records.get(user_id, []),
            # 其他相關資料...
        }

        return user_export

    def delete_user_data(self, user_id: str) -> dict:
        """
        被遺忘權 (Right to be Forgotten)
        GDPR Article 17
        """
        deleted_items = []

        # 刪除用戶資料
        if user_id in self.user_data:
            del self.user_data[user_id]
            deleted_items.append("user_data")

        # 刪除同意記錄
        if user_id in self.consent_records:
            del self.consent_records[user_id]
            deleted_items.append("consent_records")

        # 應該刪除其他所有相關資料：
        # - 日誌
        # - 分析資料
        # - 備份
        # - 快取

        return {
            "user_id": user_id,
            "deleted_at": datetime.utcnow().isoformat(),
            "deleted_items": deleted_items,
            "status": "completed"
        }

    def anonymize_old_data(self):
        """自動匿名化過期資料"""
        cutoff_date = datetime.utcnow() - timedelta(
            days=self.data_retention_days
        )

        # 匿名化超過保留期的資料
        for user_id, data in self.user_data.items():
            if data.get("created_at") < cutoff_date:
                # 匿名化而非刪除（保留統計價值）
                self._anonymize_user(user_id)

    def _anonymize_user(self, user_id: str):
        """匿名化用戶資料"""
        if user_id in self.user_data:
            # 移除所有識別資訊
            anonymized = {
                "user_id": f"anon_{hash(user_id)}",
                "anonymized_at": datetime.utcnow(),
                # 保留非識別資料用於統計
                "usage_count": self.user_data[user_id].get("usage_count"),
                # 移除其他識別資訊
            }
            self.user_data[user_id] = anonymized
```

```python
from typing import List, Dict
import re

class ContentModerator:
    """內容審核器"""

    def __init__(self):
        # 敏感詞庫（實際應該更完整）
        self.blocked_patterns = [
            r"password\s*[:=]\s*\S+",
            r"api[_-]key\s*[:=]\s*\S+",
            r"secret\s*[:=]\s*\S+",
            r"token\s*[:=]\s*\S+",
        ]

        self.sensitive_topics = [
            "violence", "hate_speech", "adult_content",
            "illegal_activity", "personal_attacks"
        ]

    def check_prompt(self, prompt: str) -> Dict:
        """檢查 prompt 是否安全"""
        issues = []

        # 檢查敏感模式
        for pattern in self.blocked_patterns:
            if re.search(pattern, prompt, re.IGNORECASE):
                issues.append({
                    "type": "sensitive_data",
                    "pattern": pattern,
                    "severity": "high"
                })

        # 檢查長度
        if len(prompt) > 10000:
            issues.append({
                "type": "excessive_length",
                "severity": "medium"
            })

        # 使用 ML 模型檢查內容（示例）
        # toxic_score = self.toxicity_model.predict(prompt)
        # if toxic_score > 0.7:
        #     issues.append({
        #         "type": "toxic_content",
        #         "score": toxic_score,
        #         "severity": "high"
        #     })

        return {
            "safe": len(issues) == 0,
            "issues": issues,
            "filtered_prompt": self.filter_prompt(prompt) if issues else prompt
        }

    def filter_prompt(self, prompt: str) -> str:
        """過濾 prompt 中的敏感內容"""
        filtered = prompt

        # 移除敏感模式
        for pattern in self.blocked_patterns:
            filtered = re.sub(
                pattern,
                "[FILTERED]",
                filtered,
                flags=re.IGNORECASE
            )

        return filtered
```

```python
class AuditLogger:

        self,
        event_type: str,
        user_id: Optional[str],
        action: str,
        resource: str,
        result: str,
        details: dict = None,
        ip_address: Optional[str] = None
    ):

            "user_id": user_id,
            "action": action,

            "details": details or {}
        }

        self,
        user_id: str,
        success: bool,
        ip_address: str,

            result="SUCCESS" if success else "FAILURE",
            ip_address=ip_address,
            details={"method": method}

        resource: str,
```
