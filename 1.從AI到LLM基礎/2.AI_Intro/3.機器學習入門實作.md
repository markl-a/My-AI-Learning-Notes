## 線性回歸

線性回歸是最基礎的機器學習算法,用於預測連續值。

### 理論基礎

目標:找到最佳的直線 `y = wx + b`,使得預測值與真實值之間的誤差最小。

損失函數(MSE): `L = (1/n) Σ(y_pred - y_true)²`

### 完整實作

```python
import numpy as np
import matplotlib.pyplot as plt

class LinearRegression:
    def __init__(self, learning_rate=0.01, n_iterations=1000):
        """
        線性回歸模型

        參數:
            learning_rate: 學習率
            n_iterations: 訓練迭代次數
        """
        self.lr = learning_rate
        self.n_iterations = n_iterations
        self.weights = None
        self.bias = None
        self.losses = []

    def fit(self, X, y):
        """訓練模型"""
        n_samples, n_features = X.shape

        # 初始化參數
        self.weights = np.zeros(n_features)
        self.bias = 0

        # 梯度下降
        for i in range(self.n_iterations):
            # 前向傳播
            y_pred = self.predict(X)

            # 計算損失
            loss = np.mean((y_pred - y) ** 2)
            self.losses.append(loss)

            # 計算梯度
            dw = (2 / n_samples) * np.dot(X.T, (y_pred - y))
            db = (2 / n_samples) * np.sum(y_pred - y)

            # 更新參數
            self.weights -= self.lr * dw
            self.bias -= self.lr * db

            if i % 100 == 0:
                print(f"Iteration {i}, Loss: {loss:.4f}")

    def predict(self, X):
        """預測"""
        return np.dot(X, self.weights) + self.bias

    def plot_losses(self):
        """繪製損失曲線"""
        plt.figure(figsize=(10, 5))
        plt.plot(self.losses)
        plt.xlabel('Iteration')
        plt.ylabel('Loss (MSE)')
        plt.title('Training Loss Over Time')
        plt.grid(True, alpha=0.3)
        plt.show()


# 生成測試數據
np.random.seed(42)
X = 2 * np.random.rand(100, 1)
y = 4 + 3 * X + np.random.randn(100, 1)

# 訓練模型
model = LinearRegression(learning_rate=0.1, n_iterations=1000)
model.fit(X, y.ravel())

# 預測
X_test = np.array([[0], [2]])
predictions = model.predict(X_test)

# 可視化結果

plt.subplot(1, 2, 1)
plt.scatter(X, y, alpha=0.5, label='Training data')

plt.xlabel('X')
plt.ylabel('y')
plt.title('Linear Regression Results')
plt.legend()
plt.grid(True, alpha=0.3)

plt.subplot(1, 2, 2)

plt.xlabel('Iteration')
plt.ylabel('Loss (MSE)')
plt.title('Training Loss')
plt.grid(True, alpha=0.3)

print(f"偏置 b: {model.bias:.4f}")
```

## 邏輯回歸

邏輯回歸用於二元分類問題。

### 理論基礎

使用 Sigmoid 函數將線性組合映射到 [0, 1] 區間:
- `σ(z) = 1 / (1 + e^(-z))`
- 損失函數: Binary Cross-Entropy

### 完整實作

```python
import numpy as np
import matplotlib.pyplot as plt

class LogisticRegression:
    def __init__(self, learning_rate=0.01, n_iterations=1000):
        self.lr = learning_rate
        self.n_iterations = n_iterations
        self.weights = None
        self.bias = None
        self.losses = []

    def sigmoid(self, z):
        """Sigmoid 激活函數"""
        return 1 / (1 + np.exp(-z))

    def fit(self, X, y):
        """訓練模型"""
        n_samples, n_features = X.shape

        # 初始化參數
        self.weights = np.zeros(n_features)
        self.bias = 0

        # 梯度下降
        for i in range(self.n_iterations):
            # 前向傳播
            linear_output = np.dot(X, self.weights) + self.bias
            y_pred = self.sigmoid(linear_output)

            # 計算損失 (Binary Cross-Entropy)
            loss = -np.mean(y * np.log(y_pred + 1e-7) +
                           (1 - y) * np.log(1 - y_pred + 1e-7))
            self.losses.append(loss)

            # 計算梯度
            dw = (1 / n_samples) * np.dot(X.T, (y_pred - y))
            db = (1 / n_samples) * np.sum(y_pred - y)

            # 更新參數
            self.weights -= self.lr * dw
            self.bias -= self.lr * db

            if i % 100 == 0:
                print(f"Iteration {i}, Loss: {loss:.4f}")

    def predict(self, X, threshold=0.5):
        """預測類別"""
        linear_output = np.dot(X, self.weights) + self.bias
        y_pred = self.sigmoid(linear_output)
        return (y_pred >= threshold).astype(int)

    def predict_proba(self, X):
        """預測概率"""
        linear_output = np.dot(X, self.weights) + self.bias
        return self.sigmoid(linear_output)


# 生成測試數據（二元分類）
np.random.seed(42)
from sklearn.datasets import make_classification

X, y = make_classification(n_samples=200, n_features=2, n_redundant=0,
                           n_informative=2, n_clusters_per_class=1,
                           random_state=42)

# 訓練模型
model = LogisticRegression(learning_rate=0.1, n_iterations=1000)
model.fit(X, y)

# 評估
predictions = model.predict(X)
accuracy = np.mean(predictions == y)
print(f"\n訓練集準確率: {accuracy:.4f}")

# 可視化決策邊界
def plot_decision_boundary(model, X, y):
    plt.figure(figsize=(10, 6))

    # 繪製數據點
    plt.scatter(X[y==0, 0], X[y==0, 1], c='blue', label='Class 0', alpha=0.5)
    plt.scatter(X[y==1, 0], X[y==1, 1], c='red', label='Class 1', alpha=0.5)

    # 繪製決策邊界
    x_min, x_max = X[:, 0].min() - 1, X[:, 0].max() + 1
    y_min, y_max = X[:, 1].min() - 1, X[:, 1].max() + 1
    xx, yy = np.meshgrid(np.linspace(x_min, x_max, 100),
                         np.linspace(y_min, y_max, 100))

    Z = model.predict(np.c_[xx.ravel(), yy.ravel()])
    Z = Z.reshape(xx.shape)

    plt.contourf(xx, yy, Z, alpha=0.2, levels=1, cmap='RdBu')
    plt.xlabel('Feature 1')
    plt.ylabel('Feature 2')
    plt.title('Logistic Regression Decision Boundary')
    plt.legend()
    plt.grid(True, alpha=0.3)
    plt.show()

plot_decision_boundary(model, X, y)
```

## K-近鄰算法

K-NN 是一種簡單而有效的分類算法,基於「物以類聚」的思想。

### 理論基礎

對於新樣本,找到訓練集中距離最近的 K 個樣本,通過投票決定類別。

### 完整實作

```python
import numpy as np
from collections import Counter
import matplotlib.pyplot as plt

class KNN:
    def __init__(self, k=3):
        """
        K-近鄰分類器

        參數:
            k: 鄰居數量
        """
        self.k = k
        self.X_train = None
        self.y_train = None

    def fit(self, X, y):
        """訓練模型（僅保存數據）"""
        self.X_train = X
        self.y_train = y

    def euclidean_distance(self, x1, x2):
        """計算歐氏距離"""
        return np.sqrt(np.sum((x1 - x2) ** 2))

    def predict(self, X):
        """預測"""
        predictions = [self._predict_single(x) for x in X]
        return np.array(predictions)

    def _predict_single(self, x):
        """預測單個樣本"""
        # 計算到所有訓練樣本的距離
        distances = [self.euclidean_distance(x, x_train)
                    for x_train in self.X_train]

        # 獲取最近的 K 個樣本的索引
        k_indices = np.argsort(distances)[:self.k]

        # 獲取這 K 個樣本的標籤
        k_nearest_labels = [self.y_train[i] for i in k_indices]

        # 投票決定類別
        most_common = Counter(k_nearest_labels).most_common(1)
        return most_common[0][0]


# 生成測試數據

# 分割訓練集和測試集

    X, y, test_size=0.2, random_state=42
```

## 決策樹

決策樹是一種樹狀結構的分類算法，通過一系列規則對數據進行劃分。

### 理論基礎

**核心概念**：
- 使用信息增益或基尼指數選擇最佳分割特徵
- 遞歸地將數據集分割成更小的子集
- 直到滿足停止條件（最大深度、最小樣本數等）

**信息熵**：`H(S) = -Σ p_i * log2(p_i)`

**信息增益**：`IG(S, A) = H(S) - Σ |S_v|/|S| * H(S_v)`

**基尼不純度**：`Gini(S) = 1 - Σ p_i²`

### 完整實作

```python
import numpy as np
from collections import Counter

class Node:
    """決策樹節點"""
    def __init__(self, feature=None, threshold=None, left=None, right=None, value=None):
        self.feature = feature      # 分割特徵的索引
        self.threshold = threshold  # 分割閾值
        self.left = left           # 左子樹
        self.right = right         # 右子樹
        self.value = value         # 葉節點的預測值

class DecisionTree:
    def __init__(self, max_depth=10, min_samples_split=2, criterion='gini'):
        """
        決策樹分類器

        參數:
            max_depth: 最大深度
            min_samples_split: 分割所需的最小樣本數
            criterion: 分割標準 ('gini' 或 'entropy')
        """
        self.max_depth = max_depth
        self.min_samples_split = min_samples_split
        self.criterion = criterion
        self.root = None

    def gini_impurity(self, y):
        """計算基尼不純度"""
        counter = Counter(y)
        impurity = 1.0
        n_samples = len(y)

        for count in counter.values():
            prob = count / n_samples
            impurity -= prob ** 2

        return impurity

    def entropy(self, y):
        """計算信息熵"""
        counter = Counter(y)
        entropy = 0.0
        n_samples = len(y)

        for count in counter.values():
            if count > 0:
                prob = count / n_samples
                entropy -= prob * np.log2(prob)

        return entropy

    def information_gain(self, y, y_left, y_right):
        """計算信息增益"""
        if self.criterion == 'gini':
            parent_impurity = self.gini_impurity(y)
            left_impurity = self.gini_impurity(y_left)
            right_impurity = self.gini_impurity(y_right)
        else:  # entropy
            parent_impurity = self.entropy(y)
            left_impurity = self.entropy(y_left)
            right_impurity = self.entropy(y_right)

        n = len(y)
        n_left = len(y_left)
        n_right = len(y_right)

        # 加權平均
        child_impurity = (n_left / n) * left_impurity + (n_right / n) * right_impurity

        # 信息增益
        return parent_impurity - child_impurity

    def split_data(self, X, y, feature, threshold):
        """根據特徵和閾值分割數據"""
        left_mask = X[:, feature] <= threshold
        right_mask = ~left_mask

        return (
            X[left_mask], y[left_mask],
            X[right_mask], y[right_mask]
        )

    def find_best_split(self, X, y):
        """找到最佳分割"""
        best_gain = -1
        best_feature = None
        best_threshold = None

        n_features = X.shape[1]

        # 遍歷所有特徵
        for feature in range(n_features):
            # 獲取該特徵的所有唯一值作為候選閾值
            thresholds = np.unique(X[:, feature])

            # 遍歷所有閾值
            for threshold in thresholds:
                # 分割數據
                X_left, y_left, X_right, y_right = self.split_data(X, y, feature, threshold)

                # 如果分割後任一子集為空，跳過
                if len(y_left) == 0 or len(y_right) == 0:
                    continue

                # 計算信息增益
                gain = self.information_gain(y, y_left, y_right)

                # 更新最佳分割
                if gain > best_gain:
                    best_gain = gain
                    best_feature = feature
                    best_threshold = threshold

        return best_feature, best_threshold, best_gain

    def build_tree(self, X, y, depth=0):
        """遞歸構建決策樹"""
        n_samples, n_features = X.shape
        n_labels = len(np.unique(y))

        # 停止條件
        if (depth >= self.max_depth or
            n_samples < self.min_samples_split or
            n_labels == 1):
            # 創建葉節點
            leaf_value = Counter(y).most_common(1)[0][0]
            return Node(value=leaf_value)

        # 找到最佳分割
        best_feature, best_threshold, best_gain = self.find_best_split(X, y)

        # 如果沒有找到有效的分割
        if best_feature is None:
            leaf_value = Counter(y).most_common(1)[0][0]
            return Node(value=leaf_value)

        # 分割數據
        X_left, y_left, X_right, y_right = self.split_data(
            X, y, best_feature, best_threshold
        )

        # 遞歸構建左右子樹
        left_subtree = self.build_tree(X_left, y_left, depth + 1)
        right_subtree = self.build_tree(X_right, y_right, depth + 1)

        # 返回決策節點
        return Node(
            feature=best_feature,
            threshold=best_threshold,
            left=left_subtree,
            right=right_subtree
        )

    def fit(self, X, y):
        """訓練決策樹"""
        self.root = self.build_tree(X, y)
        return self

    def _predict_single(self, x, node):
        """預測單個樣本"""
        # 如果是葉節點，返回預測值
        if node.value is not None:
            return node.value

        # 根據特徵值決定走左子樹還是右子樹
        if x[node.feature] <= node.threshold:
            return self._predict_single(x, node.left)
        else:
            return self._predict_single(x, node.right)

    def predict(self, X):
        """預測"""
        return np.array([self._predict_single(x, self.root) for x in X])

    def print_tree(self, node=None, depth=0):
        """打印決策樹結構"""
        if node is None:
            node = self.root

        indent = "  " * depth

        if node.value is not None:
            print(f"{indent}Leaf: class = {node.value}")
        else:
            print(f"{indent}Feature {node.feature} <= {node.threshold:.4f}")
            print(f"{indent}Left:")
            self.print_tree(node.left, depth + 1)
            print(f"{indent}Right:")
            self.print_tree(node.right, depth + 1)


# 生成測試數據

    ax = axes[idx]
```

### 決策樹的優缺點

**優點**：
- ✅ 易於理解和解釋（可視化）
- ✅ 不需要數據標準化
- ✅ 可以處理數值和類別特徵
- ✅ 能夠處理非線性關係

**缺點**：
- ❌ 容易過擬合（需要剪枝）
- ❌ 對訓練數據的小變化敏感
- ❌ 可能創建偏斜的樹
- ❌ 在某些問題上性能不如其他算法

### 改進方向

1. **剪枝**：後剪枝（post-pruning）減少過擬合
2. **集成方法**：隨機森林（Random Forest）
3. **提升方法**：梯度提升樹（Gradient Boosting）
4. **特徵重要性**：分析哪些特徵最重要

## K-Means 聚類

K-Means 是最常用的無監督學習算法之一。

### 理論基礎

1. 隨機初始化 K 個聚類中心
2. 將每個樣本分配到最近的聚類中心
3. 更新聚類中心為該類所有樣本的均值
4. 重複步驟 2-3 直到收斂

### 完整實作

```python
import numpy as np
import matplotlib.pyplot as plt

class KMeans:
    def __init__(self, n_clusters=3, max_iters=100):
        """
        K-Means 聚類算法

        參數:
            n_clusters: 聚類數量
            max_iters: 最大迭代次數
        """
        self.n_clusters = n_clusters
        self.max_iters = max_iters
        self.centroids = None
        self.labels = None

    def fit(self, X):
        """訓練模型"""
        n_samples, n_features = X.shape

        # 隨機初始化聚類中心
        random_indices = np.random.choice(n_samples, self.n_clusters, replace=False)
        self.centroids = X[random_indices]

        # 迭代優化
        for iteration in range(self.max_iters):
            # 1. 分配樣本到最近的聚類中心
            self.labels = self._assign_clusters(X)

            # 保存舊的聚類中心
            old_centroids = self.centroids.copy()

            # 2. 更新聚類中心
            self.centroids = self._update_centroids(X)

            # 檢查是否收斂
            if np.allclose(old_centroids, self.centroids):
                print(f"收斂於第 {iteration + 1} 次迭代")
                break

        return self

    def _assign_clusters(self, X):
        """將每個樣本分配到最近的聚類中心"""
        distances = np.sqrt(((X - self.centroids[:, np.newaxis])**2).sum(axis=2))
        return np.argmin(distances, axis=0)

    def _update_centroids(self, X):
        """更新聚類中心"""
        centroids = np.zeros((self.n_clusters, X.shape[1]))
        for i in range(self.n_clusters):
            cluster_points = X[self.labels == i]
            if len(cluster_points) > 0:
                centroids[i] = cluster_points.mean(axis=0)
        return centroids

    def predict(self, X):
        """預測新樣本的聚類"""
        return self._assign_clusters(X)

    def inertia(self, X):
        """計算慣性（樣本到聚類中心的距離平方和）"""
        total_distance = 0
        for i in range(self.n_clusters):
            cluster_points = X[self.labels == i]
            if len(cluster_points) > 0:
                distances = np.sqrt(((cluster_points - self.centroids[i])**2).sum(axis=1))
                total_distance += (distances ** 2).sum()
        return total_distance
```

## 簡單神經網路

實作一個簡單的前饋神經網路用於分類。

### 網路結構

輸入層 -> 隱藏層(ReLU) -> 輸出層(Softmax)

### 完整實作

```python
import numpy as np
import matplotlib.pyplot as plt

class SimpleNeuralNetwork:
    def __init__(self, input_size, hidden_size, output_size, learning_rate=0.01):
        """
        簡單的神經網路

        參數:
            input_size: 輸入層大小
            hidden_size: 隱藏層大小
            output_size: 輸出層大小
            learning_rate: 學習率
        """
        self.lr = learning_rate

        # 初始化權重
        self.W1 = np.random.randn(input_size, hidden_size) * 0.01
        self.b1 = np.zeros((1, hidden_size))
        self.W2 = np.random.randn(hidden_size, output_size) * 0.01
        self.b2 = np.zeros((1, output_size))

        self.losses = []
        self.accuracies = []

    def relu(self, Z):
        """ReLU 激活函數"""
        return np.maximum(0, Z)

    def relu_derivative(self, Z):
        """ReLU 導數"""
        return (Z > 0).astype(float)

    def softmax(self, Z):
        """Softmax 函數"""
        exp_Z = np.exp(Z - np.max(Z, axis=1, keepdims=True))
        return exp_Z / np.sum(exp_Z, axis=1, keepdims=True)

    def forward(self, X):
        """前向傳播"""
        # 隱藏層
        self.Z1 = X @ self.W1 + self.b1
        self.A1 = self.relu(self.Z1)

        # 輸出層
        self.Z2 = self.A1 @ self.W2 + self.b2
        self.A2 = self.softmax(self.Z2)

        return self.A2

    def backward(self, X, y, y_pred):
        """反向傳播"""
        m = X.shape[0]

        # 輸出層梯度
        dZ2 = y_pred - y
        dW2 = (1/m) * (self.A1.T @ dZ2)
        db2 = (1/m) * np.sum(dZ2, axis=0, keepdims=True)

        # 隱藏層梯度
        dA1 = dZ2 @ self.W2.T
        dZ1 = dA1 * self.relu_derivative(self.Z1)
        dW1 = (1/m) * (X.T @ dZ1)
        db1 = (1/m) * np.sum(dZ1, axis=0, keepdims=True)

        # 更新權重
        self.W1 -= self.lr * dW1
        self.b1 -= self.lr * db1
        self.W2 -= self.lr * dW2
        self.b2 -= self.lr * db2

    def compute_loss(self, y_true, y_pred):
        """計算交叉熵損失"""
        m = y_true.shape[0]
        loss = -np.sum(y_true * np.log(y_pred + 1e-7)) / m
        return loss

    def train(self, X, y, epochs=1000, verbose=True):
        """訓練模型"""
        for epoch in range(epochs):
            # 前向傳播
            y_pred = self.forward(X)

            # 計算損失
            loss = self.compute_loss(y, y_pred)
            self.losses.append(loss)

            # 計算準確率
            predictions = np.argmax(y_pred, axis=1)
            true_labels = np.argmax(y, axis=1)
            accuracy = np.mean(predictions == true_labels)
            self.accuracies.append(accuracy)

            # 反向傳播
            self.backward(X, y, y_pred)

            if verbose and epoch % 100 == 0:
                print(f"Epoch {epoch}, Loss: {loss:.4f}, Accuracy: {accuracy:.4f}")

    def predict(self, X):
        """預測"""
        y_pred = self.forward(X)
        return np.argmax(y_pred, axis=1)

    def plot_training_history(self):
        """繪製訓練歷史"""
        plt.figure(figsize=(12, 4))

        plt.subplot(1, 2, 1)
        plt.plot(self.losses)
        plt.xlabel('Epoch')
        plt.ylabel('Loss')
        plt.title('Training Loss')
        plt.grid(True, alpha=0.3)

        plt.subplot(1, 2, 2)
        plt.plot(self.accuracies)
        plt.xlabel('Epoch')
        plt.ylabel('Accuracy')
        plt.title('Training Accuracy')
        plt.grid(True, alpha=0.3)

        plt.tight_layout()
        plt.show()


# 生成測試數據
```
