## 為什麼 AI 需要 NumPy

在 AI 和機器學習中，我們經常需要處理大量的資料和複雜的數學運算：

- **向量和矩陣運算**：神經網路的權重、啟動值都是矩陣
- **高效計算**：NumPy 使用 C 語言實現，比 Python 原生串列快 10-100 倍
- **廣播機制**：簡化不同形狀陣列間的運算
- **豐富的數學函式**：線性代數、統計、傅立葉轉換等

**效能比較範例**：
```python
import numpy as np
import time

# Python 原生串列
python_list = list(range(1000000))
start = time.time()
result = [x * 2 for x in python_list]
python_time = time.time() - start

# NumPy 陣列
numpy_array = np.arange(1000000)
start = time.time()
result = numpy_array * 2
numpy_time = time.time() - start

print(f"Python 串列時間: {python_time:.4f} 秒")
print(f"NumPy 陣列時間: {numpy_time:.4f} 秒")
print(f"NumPy 快了 {python_time/numpy_time:.1f} 倍")
```

### 矩陣運算（線性代數）

```python
import numpy as np

# 矩陣乘法（點積）
A = np.array([[1, 2], [3, 4]])
B = np.array([[5, 6], [7, 8]])

print("矩陣 A:\n", A)
print("\n矩陣 B:\n", B)

# 元素級乘法
print("\n元素級乘法 A * B:\n", A * B)

# 矩陣乘法
print("\n矩陣乘法 A @ B:\n", A @ B)
# 或使用 np.dot(A, B)

# 向量點積
v1 = np.array([1, 2, 3])
v2 = np.array([4, 5, 6])
print("\n向量點積:", np.dot(v1, v2))  # 1*4 + 2*5 + 3*6 = 32

# 矩陣的逆
matrix = np.array([[1, 2], [3, 4]])
inv_matrix = np.linalg.inv(matrix)
print("\n矩陣的逆:\n", inv_matrix)

# 驗證：A @ A^(-1) = I
print("\n驗證 A @ A^(-1):\n", matrix @ inv_matrix)

# 特徵值和特徵向量
eigenvalues, eigenvectors = np.linalg.eig(matrix)
print("\n特徵值:", eigenvalues)
print("特徵向量:\n", eigenvectors)
```

### 1. 資料預處理

```python
import numpy as np

# 模擬一個資料集
np.random.seed(42)
data = np.random.randn(100, 5) * 10 + 50  # 100 個樣本，5 個特徵

# 歸一化（Min-Max scaling）到 [0, 1]
min_val = data.min(axis=0)
max_val = data.max(axis=0)
normalized_data = (data - min_val) / (max_val - min_val)

print("\n歸一化後的範圍:")
print("最小值:", normalized_data.min(axis=0))
print("最大值:", normalized_data.max(axis=0))
```

### 2. 實作簡單的線性回歸

```python
import numpy as np
import matplotlib.pyplot as plt

# 生成訓練資料
np.random.seed(42)
X = np.random.rand(100, 1) * 10  # 100 個樣本
y = 3 * X + 7 + np.random.randn(100, 1) * 2  # y = 3x + 7 + noise

# 添加偏置項（intercept）
X_b = np.c_[np.ones((100, 1)), X]  # 在 X 前面加一列 1

# 使用正規方程求解：θ = (X^T X)^(-1) X^T y
theta_best = np.linalg.inv(X_b.T @ X_b) @ X_b.T @ y

print("最佳參數 θ:")
print(f"截距: {theta_best[0][0]:.2f}")
print(f"斜率: {theta_best[1][0]:.2f}")

# 預測
X_new = np.array([[0], [10]])
X_new_b = np.c_[np.ones((2, 1)), X_new]
y_predict = X_new_b @ theta_best

# 可視化
plt.figure(figsize=(10, 6))
plt.scatter(X, y, alpha=0.5, label='Training data')
plt.plot(X_new, y_predict, 'r-', linewidth=2, label='Predictions')
plt.xlabel('X')
plt.ylabel('y')
plt.title('Linear Regression with NumPy')
plt.legend()
plt.grid(True, alpha=0.3)
plt.show()

print(f"\n預測 X=0 時: y={y_predict[0][0]:.2f}")
print(f"預測 X=10 時: y={y_predict[1][0]:.2f}")
```

### 3. 實作 Sigmoid 激活函式

```python
import numpy as np
import matplotlib.pyplot as plt

def sigmoid(x):
    """Sigmoid 激活函式"""
    return 1 / (1 + np.exp(-x))

def sigmoid_derivative(x):
    """Sigmoid 導數"""
    s = sigmoid(x)
    return s * (1 - s)

# 生成資料
x = np.linspace(-10, 10, 100)
y_sigmoid = sigmoid(x)
y_derivative = sigmoid_derivative(x)

# 可視化
plt.figure(figsize=(12, 5))

plt.subplot(1, 2, 1)
plt.plot(x, y_sigmoid, 'b-', linewidth=2)
plt.grid(True, alpha=0.3)
plt.xlabel('x')
plt.ylabel('sigmoid(x)')
plt.title('Sigmoid Activation Function')
plt.axhline(y=0.5, color='r', linestyle='--', alpha=0.5)
plt.axvline(x=0, color='r', linestyle='--', alpha=0.5)

plt.subplot(1, 2, 2)
plt.plot(x, y_derivative, 'g-', linewidth=2)
plt.grid(True, alpha=0.3)
plt.xlabel('x')
plt.ylabel("sigmoid'(x)")
plt.title('Sigmoid Derivative')

plt.tight_layout()
plt.show()

# 實際應用：二元分類
z = np.array([-2, -1, 0, 1, 2])
probabilities = sigmoid(z)
predictions = (probabilities >= 0.5).astype(int)
```

### 4. 實作簡單的神經網路前向傳播

```python
import numpy as np

def relu(x):
    """ReLU 激活函式"""
    return np.maximum(0, x)

def softmax(x):
    """Softmax 函數（用於多分類）"""
    exp_x = np.exp(x - np.max(x, axis=1, keepdims=True))
    return exp_x / np.sum(exp_x, axis=1, keepdims=True)

# 初始化一個簡單的神經網路
np.random.seed(42)

# 網路結構：輸入層(4) -> 隱藏層(5) -> 輸出層(3)
input_size = 4
hidden_size = 5
output_size = 3

# 隨機初始化權重和偏置
W1 = np.random.randn(input_size, hidden_size) * 0.01
b1 = np.zeros((1, hidden_size))
W2 = np.random.randn(hidden_size, output_size) * 0.01
b2 = np.zeros((1, output_size))

# 前向傳播
def forward_propagation(X, W1, b1, W2, b2):
    """
    X: 輸入資料 (batch_size, input_size)
    返回: 輸出概率分布
    """
    # 第一層
    Z1 = X @ W1 + b1
    A1 = relu(Z1)

    # 第二層（輸出層）
    Z2 = A1 @ W2 + b2
    A2 = softmax(Z2)

    return A2

# 測試
X_test = np.random.randn(10, input_size)  # 10 個樣本
predictions = forward_propagation(X_test, W1, b1, W2, b2)

print("前向傳播結果:")
print("輸入形狀:", X_test.shape)
print("輸出形狀:", predictions.shape)
print("\n前 3 個樣本的預測概率:")
print(predictions[:3])
print("\n預測類別:")
print(np.argmax(predictions[:3], axis=1))
print("\n每行概率和（應該都是 1）:")
print(predictions.sum(axis=1)[:5])
```

## 進階技巧

### 1. 向量化技巧

向量化是 NumPy 效能優化的關鍵，避免使用 Python 迴圈。

```python
import numpy as np
import time

# 範例：計算兩個向量的歐幾里得距離

# 方法 1：使用 Python 迴圈（慢）
def euclidean_distance_loop(a, b):
    distance = 0
    for i in range(len(a)):
        distance += (a[i] - b[i]) ** 2
    return np.sqrt(distance)

# 方法 2：使用 NumPy 向量化（快）
def euclidean_distance_vectorized(a, b):
    return np.sqrt(np.sum((a - b) ** 2))

# 效能比較
a = np.random.rand(10000)
b = np.random.rand(10000)

start = time.time()
for _ in range(1000):
    result1 = euclidean_distance_loop(a, b)
loop_time = time.time() - start

start = time.time()
for _ in range(1000):
    result2 = euclidean_distance_vectorized(a, b)
vectorized_time = time.time() - start

print(f"迴圈方法: {loop_time:.4f} 秒")
print(f"向量化方法: {vectorized_time:.4f} 秒")
print(f"向量化快了 {loop_time/vectorized_time:.1f} 倍")
```

### 3. 高級索引技巧

```python
import numpy as np
```

```python
def moving_average(data, window):
    """計算簡單移動平均（SMA）"""
    cumsum = np.cumsum(np.insert(data, 0, 0))
    return (cumsum[window:] - cumsum[:-window]) / window

ma5 = moving_average(prices, 5)    # 5日均線
ma20 = moving_average(prices, 20)  # 20日均線
ma60 = moving_average(prices, 60)  # 季線

# 需要對齊陣列長度
ma5_aligned = ma5[15:]  # 跳過前15天
ma20_aligned = ma20

# 檢測交叉點
golden_cross = np.where(
    (ma5_aligned[:-1] < ma20_aligned[:-1]) &  # 前一天 5MA 低於 20MA
    (ma5_aligned[1:] > ma20_aligned[1:])      # 當天 5MA 高於 20MA
)[0] + 20  # 調整索引

print("=== 技術分析結果 ===")
print(f"期間: {days} 個交易日")
print(f"起始價: {prices[0]:.2f} 元")

print(f"最高價: {prices.max():.2f} 元 (第 {prices.argmax()+1} 天)")
```

```python
import numpy as np

# 模擬醫療影像資料（例如 CT 掃描）
np.random.seed(42)

# 建立模擬影像（256x256，灰階）
image = np.random.randint(0, 256, (256, 256), dtype=np.uint8)

# 加入高斯雜訊
noise = np.random.normal(0, 15, image.shape)
noisy_image = np.clip(image + noise, 0, 255).astype(np.uint8)

# 簡單的中值濾波去雜訊
def median_filter(img, kernel_size=3):
    """簡單的中值濾波實作"""
    pad = kernel_size // 2
    filtered = np.zeros_like(img)

    for i in range(pad, img.shape[0] - pad):
        for j in range(pad, img.shape[1] - pad):
            window = img[i-pad:i+pad+1, j-pad:j+pad+1]
            filtered[i, j] = np.median(window)

    return filtered

filtered_image = median_filter(noisy_image, kernel_size=3)

# 計算雜訊移除效果
mse_before = np.mean((noisy_image.astype(float) - image.astype(float)) ** 2)
mse_after = np.mean((filtered_image.astype(float) - image.astype(float)) ** 2)

print("=== 影像處理結果 ===")
print(f"原始影像大小: {image.shape}")
print(f"雜訊影像 MSE: {mse_before:.2f}")
print(f"濾波後 MSE: {mse_after:.2f}")
print(f"改善率: {(1 - mse_after/mse_before) * 100:.2f}%")

# 影像對比度增強（直方圖等化）
def histogram_equalization(img):
    """簡單的直方圖等化"""
    hist, bins = np.histogram(img.flatten(), 256, [0, 256])
    cdf = hist.cumsum()
    cdf_normalized = cdf * hist.max() / cdf.max()

    # 建立映射表
    cdf_m = np.ma.masked_equal(cdf, 0)
    cdf_m = (cdf_m - cdf_m.min()) * 255 / (cdf_m.max() - cdf_m.min())
    cdf = np.ma.filled(cdf_m, 0).astype('uint8')

    # 應用映射
    return cdf[img]

enhanced_image = histogram_equalization(image)
print(f"\n對比度增強完成")
print(f"原始影像對比度（標準差）: {image.std():.2f}")
print(f"增強後對比度（標準差）: {enhanced_image.std():.2f}")
```

## 從 NumPy 到深度學習框架

### NumPy 與深度學習框架的關係

深度學習框架（TensorFlow、PyTorch）在底層都使用了類似 NumPy 的張量（Tensor）操作，但增加了自動微分、GPU 加速等功能。

**主要差異**：
| 特性 | NumPy | PyTorch | TensorFlow |
|------|-------|---------|------------|
| 資料結構 | ndarray | Tensor | Tensor |
| 自動微分 | ❌ | ✅ | ✅ |
| GPU 支援 | ❌ | ✅ | ✅ |
| 靜態圖 | N/A | ❌ | ✅ (選擇性) |
| 語法相似度 | 基準 | 95% | 80% |

### 1. NumPy ↔ PyTorch 互轉

### 3. 使用 PyTorch 實作神經網路（從 NumPy 過渡）

```python
import torch
import torch.nn as nn
import torch.optim as optim
import numpy as np

# 使用 NumPy 生成資料
np.random.seed(42)
X_numpy = np.random.randn(1000, 10).astype(np.float32)
y_numpy = np.random.randint(0, 2, (1000,)).astype(np.int64)

# 轉換為 PyTorch Tensor
X = torch.from_numpy(X_numpy)
y = torch.from_numpy(y_numpy)

# 定義神經網路
class SimpleNN(nn.Module):
    def __init__(self, input_size, hidden_size, output_size):
        super(SimpleNN, self).__init__()
        self.fc1 = nn.Linear(input_size, hidden_size)
        self.relu = nn.ReLU()
        self.fc2 = nn.Linear(hidden_size, output_size)

    def forward(self, x):
        x = self.fc1(x)
        x = self.relu(x)
        x = self.fc2(x)
        return x

# 初始化模型
model = SimpleNN(input_size=10, hidden_size=20, output_size=2)
criterion = nn.CrossEntropyLoss()
optimizer = optim.Adam(model.parameters(), lr=0.001)

# 訓練模型
n_epochs = 50
for epoch in range(n_epochs):
    # 前向傳播
    outputs = model(X)
    loss = criterion(outputs, y)

    # 反向傳播
    optimizer.zero_grad()
    loss.backward()
    optimizer.step()

    if (epoch + 1) % 10 == 0:
        print(f'Epoch [{epoch+1}/{n_epochs}], Loss: {loss.item():.4f}')

# 預測
with torch.no_grad():
    predictions = model(X)
    predicted_classes = torch.argmax(predictions, dim=1)
    accuracy = (predicted_classes == y).float().mean()
    print(f'\n訓練集準確率: {accuracy:.4f}')
```

### 5. 實作自定義梯度計算

**NumPy 手動計算梯度**：
```python
import numpy as np

def sigmoid(x):
    return 1 / (1 + np.exp(-x))

def sigmoid_derivative(x):
    s = sigmoid(x)
    return s * (1 - s)

# 前向傳播
x = np.array([0.5])
w = np.array([2.0])
b = 0.3

z = x * w + b
a = sigmoid(z)

# 手動計算梯度
dL_da = 1  # 假設損失對輸出的梯度為 1
da_dz = sigmoid_derivative(z)
dz_dw = x
dz_db = 1

dL_dw = dL_da * da_dz * dz_dw
dL_db = dL_da * da_dz * dz_db

print(f"手動計算的梯度: dL/dw = {dL_dw}, dL/db = {dL_db}")
```

**PyTorch 自動微分**：
```python
import torch

# 定義需要梯度的變數
x = torch.tensor([0.5], requires_grad=False)
w = torch.tensor([2.0], requires_grad=True)
b = torch.tensor([0.3], requires_grad=True)

# 前向傳播
z = x * w + b
a = torch.sigmoid(z)

# 反向傳播（自動計算梯度）
a.backward()

print(f"PyTorch 自動微分: dL/dw = {w.grad}, dL/db = {b.grad}")
```

### 8. 從 NumPy 實作過渡到 PyTorch 實作

**NumPy 版本的線性回歸**：
```python
import numpy as np

class LinearRegressionNumPy:
    def __init__(self, lr=0.01):
        self.lr = lr
        self.w = None
        self.b = None

    def fit(self, X, y, epochs=100):
        n_samples, n_features = X.shape
        self.w = np.zeros(n_features)
        self.b = 0

        for _ in range(epochs):
            y_pred = X @ self.w + self.b
            dw = (2/n_samples) * X.T @ (y_pred - y)
            db = (2/n_samples) * np.sum(y_pred - y)

            self.w -= self.lr * dw
            self.b -= self.lr * db

    def predict(self, X):
        return X @ self.w + self.b
```

**PyTorch 版本的線性回歸**：
```python
import torch
import torch.nn as nn

class LinearRegressionPyTorch(nn.Module):
    def __init__(self, input_size):
        super().__init__()
        self.linear = nn.Linear(input_size, 1)

    def forward(self, x):
        return self.linear(x)

# 使用範例
X = torch.randn(100, 5)
y = torch.randn(100, 1)

model = LinearRegressionPyTorch(input_size=5)

    outputs = model(X)
    loss = criterion(outputs, y)

    optimizer.zero_grad()
    loss.backward()
    optimizer.step()
```

| 操作 | NumPy | PyTorch | TensorFlow |
|------|-------|---------|------------|
| 建立陣列 | `np.array([1,2,3])` | `torch.tensor([1,2,3])` | `tf.constant([1,2,3])` |
